Public Function OrdenModelo(QueIdOrd As Variant) As Variant
Dim Dbs As Database
Dim Rst As Recordset
    Set Dbs = CurrentDb()
    Set Rst = Dbs.OpenRecordset("SELECT Ordenes.IdOrdenes, Modelos.Modelo FROM Ordenes INNER JOIN Modelos ON Ordenes.IdModelos = Modelos.IdModelos WHERE (Ordenes.IdOrdenes)=" & QueIdOrd)
       OrdenModelo = Rst!Modelo

    Dbs.Close
End Function

Public Function ConsumoOrd(Ord As Integer) As Single
Dim Dbs As Database
Dim Rst As Recordset
    Set Dbs = CurrentDb()
    Set Rst = Dbs.OpenRecordset("SELECT Salidas.IdOrdenes, Sum(([telasal1]+[telasal2])/[cantcorte]) AS Consumo FROM CantidadCorte INNER JOIN (SalidasDet INNER JOIN Salidas ON SalidasDet.IdSalidas = Salidas.IdSalidas) ON CantidadCorte.IdOrdenes = Salidas.IdOrdenes GROUP BY Salidas.IdOrdenes, CantidadCorte.CantCorte HAVING ((CantidadCorte.CantCorte)<>0) AND salidas.idordenes=" & Ord)
 
      ConsumoOrd = Rst!Consumo
    
    Rst.Close
    Dbs.Close

End Function

Public Function SacarUPC(CodigoEmp As String, Model As String) As String
Dim Texto1 As String
Texto1 = CodigoEmp + Model

If Len(Texto1) <> 12 Or IsNull(Texto1) = True Then
MsgBox "Debes de meter 12 digitos"
Exit Function
Else

Dim Total, Digito, Resultado, Resto, RestoDun, ResultadoDun As Integer
Dim Contador As Single

For Contador = 1 To 12
  Digito = Mid(Texto1, Contador, 1)
    If Contador Mod 2 = 0 Then
       Total = Total + (Digito * 3)
       Else
       Total = Total + (Digito * 1)
    End If
    
Next

Resto = Total Mod 10
RestoDun = (Total + 3) Mod 10

If Resto <> 0 Then
Resultado = 10 - Resto
End If

If RestoDun <> 0 Then
ResultadoDun = 10 - RestoDun
End If

SacarUPC = Texto1 + "" + Str(Resultado)
'CodigoDUN = "1 " + Texto1 + "" + Str(ResultadoDun)

End If
End Function
