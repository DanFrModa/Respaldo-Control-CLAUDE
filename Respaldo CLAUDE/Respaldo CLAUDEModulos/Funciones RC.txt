Public Function DiasLaborables(FechaDesde As Date, FechaHasta As Date) As Integer
Dim Laborables As Integer
On Error GoTo DiasLaborables_Error

Laborables = 1 + (FechaHasta - FechaDesde)

Laborables = Laborables - DateDiff("ww", FechaDesde, FechaHasta, vbSaturday)
Laborables = Laborables - DateDiff("ww", FechaDesde, FechaHasta, vbSunday)
Laborables = Laborables + (WeekDay(FechaDesde, vbSunday) = 1)
Laborables = Laborables + (WeekDay(FechaDesde, vbSaturday) = 1)
DiasLaborables = Laborables


DiasLaborables_Error:

MsgBox Err.Description
End Function




Public Function FechaHabiles(FechaUltima As Date, NumDias As Integer) As Date
Dim QueDia, CuantasSem As Integer
Dim CuantDias As Single


CuantasSem = (NumDias + 1) / 5
CuantDias = (NumDias / 5) - CuantasSem
CuantDias = CuantDias * 10
CuantDias = CuantDias / 2

QueDia = (CuantasSem * 7) + CuantDias

FechaHabiles = FechaUltima - QueDia


End Function

Public Function CuantosSabYDom(FechaDesde As Date, FechaHasta As Date) As Integer
Dim Sab, Dom As Integer
Sab = DateDiff("ww", FechaDesde, FechaHasta, vbSaturday)
Dom = DateDiff("ww", FechaDesde, FechaHasta, vbSunday)

CuantosSabYDom = Sab + Dom

End Function

Public Function TiempoPorArticulo(IdArt As Integer) As Integer
Dim Dbs As Database
Dim pp As Recordset

Set Dbs = CurrentDb
Set pp = Dbs.OpenRecordset("SELECT CP_TiempoArticulos.IdCP_Articulos, CP_TiempoArticulos.TiempoTotal FROM CP_TiempoArticulos WHERE CP_TiempoArticulos.IdCP_Articulos= " & IdArt)

TiempoPorArticulo = pp!TiempoTotal + TiempoColchonCostura()
pp.Close
Dbs.Close
End Function

Dim Dbs As Database

Else

Public Function NumeroAString(Numero As Integer, Proc As String) As String
Dim a As String
a = Numero
If Len(a) = 1 Then
 a = "0" + a
End If
NumeroAString = a + " - " + Proc
End Function
