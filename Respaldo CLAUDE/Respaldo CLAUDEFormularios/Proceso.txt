Private Sub CancelarOrd_Click()
If MsgBox("Estas seguro de querer cancelar la orden " & Numero & "?", vbYesNo, "CANCELAR ORDEN") = vbYes Then
Dim Dbs As Database
Set Dbs = CurrentDb()
 Dbs.Execute ("UPDATE Ordenes SET OrdCancelada = True WHERE IdOrdenes = " & IdOrdenes)
Dbs.Close
Salir.SetFocus
CancelarOrd.Visible = False
CancelEtiqueta.Visible = True
End If
End Sub

End If

Private Sub QueModelo_AfterUpdate()
Dim a As Integer
DoCmd.ApplyFilter , "[Proceso]![idmodelos]=forms![Proceso]![quemodelo]"
Me.RecordsetClone.MoveLast
a = Me.RecordsetClone.RecordCount
Me!NumeroOrdenes = a
If a > 1 Then
   Sig.Visible = True
   Ant.Visible = True
   Else
   Sig.Visible = False
   Ant.Visible = False
End If
End Sub
