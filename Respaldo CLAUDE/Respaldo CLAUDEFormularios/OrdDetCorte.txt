Private Sub Copiar_Click()
On Error GoTo errores
Dim i, TtAct As Integer
For i = 1 To 8
 DoCmd.GoToRecord , , acPrevious
  TtAct = Me("T" & i)
 DoCmd.GoToRecord , , acNext
 Me("T" & i) = TtAct
Next i
Exit Sub

errores:
MsgBox Err.Description
End Sub




Private Sub Form_BeforeInsert(Cancel As Integer)
IdCorte = QueIdCorte
End Sub

Private Sub Form_Close()
''Me.Recalc
''Forms![Proceso]![ProcesoCorte].Form![Cantidad] = Tot
On Error GoTo errores
Dim Rst As Recordset
Me.Recalc

Set Rst = CurrentDb.OpenRecordset("SELECT IdCorte, Sum([TC1]+[tc2]+[tc3]+[tc4]+[tc5]+[tc6]+[tc7]+[tc8]) AS Total FROM OrdenesDetCorte GROUP BY IdCorte HAVING (IdCorte)=" & IdCorte)

If IsNull(Rst!Total) = True Or (Rst.EOF And Rst.BOF) Then
CurrentDb.Execute "UPDATE Corte SET Cantidad=0 WHERE IdCorte=" & IdCorte
Else
CurrentDb.Execute "UPDATE Corte SET Cantidad=" & Rst!Total & " WHERE IdCorte=" & IdCorte

End If
Rst.Close

errores:
If Err.Number = 3021 Then

End If

If EstaCargado("Proceso") = True Then

Forms![Proceso]![ProcesoCorte].Form.RecordsetClone.FindFirst "[IdCorte] = " & IdCorte
   Forms![Proceso]![ProcesoCorte].Form.Bookmark = Forms![Proceso]![ProcesoCorte].Form.RecordsetClone.Bookmark
End If

End Sub

Public Sub BorrarAa()
If Len(TallasLoc) < 16 Then
Dim i, Tam, Tt As Integer
Tam = Len(TallasLoc) + 2
For i = 16 To Tam Step -2
Tt = i / 2
Me("T" & Tt).Visible = False
Me("T" & Tt & " Etiqueta").Visible = False
Next i
End If

If Len(TallasLoc) > 4 Then
Dim j, Tam1, Tt1, Di As Integer
 Di = Len(TallasLoc)
 For j = 4 To Di Step 2
 Tt1 = j / 2
 Me("T" & Tt1).Visible = True
 Me("T" & Tt1 & " Etiqueta").Visible = True
Next j
End If
End Sub
