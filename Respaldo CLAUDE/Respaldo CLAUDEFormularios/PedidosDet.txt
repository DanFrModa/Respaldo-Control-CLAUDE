Private Sub CopiarDetPed()
Dim a As Integer
Dim DetPed As Recordset
Dim strmensaje As String
Dim Dbs As Database
Set Dbs = CurrentDb()
a = Forms![pedidos]![IdPedidosAnt]
Set DetPed = Dbs.OpenRecordset("SELECT IdPedidos, PedidosDet.IdModelos, CantPed, Precio, Modelos.Modelo FROM PedidosDet INNER JOIN Modelos ON PedidosDet.IdModelos = Modelos.IdModelos WHERE (IdPedidos)=" & a)

With DetPed
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             
             
            strmensaje = "Modelo: " & !Modelo & vbCr & "Cantidad: " & !CantPed & vbCr & "Precio: " & !Precio & vbCr & vbCr & _
                "(Habilitacion " & (.AbsolutePosition + 1) & _
                " de " & .RecordCount & ")"
            If MsgBox(strmensaje, vbOKCancel, "COPIAR PEDIDOS") = vbCancel _
                Then GoTo OtroMas
        
            Me![IdPedidos] = Forms![pedidos]![IdPedidos]
            Me![IdModelos] = !IdModelos
            Me![CantPed] = !CantPed
            Me![Precio] = !Precio
           
OtroMas:
            DoCmd.GoToRecord , , acNewRec
        
            .MoveNext

        Loop

        .Close
    End With
Dbs.Close
End Sub

Private Sub CantFalt_GotFocus()
If IsNull(CantFalt) = True Or CantFalt = 0 Then
 CantFalt = CantPed
End If
End Sub

Private Sub CopiarMod_Click()
Call CopiarDetPed
