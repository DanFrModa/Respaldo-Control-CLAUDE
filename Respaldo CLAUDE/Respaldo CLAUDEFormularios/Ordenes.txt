Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdOrdenes, IdMaquileros, IdEtiquetasM, IdClientes, IdEmpresas, Tallas, FechaEntregaRC, IdTelasDis, Composicion, CompForzada FROM Ordenes where idordenes=" & a)

Public Sub OcultarTallas()
If Len([Tallas]) < 16 Then
Dim i, Tam, Tt As Integer
Tam = Len([Tallas]) + 2
For i = 16 To Tam Step -2
Tt = i / 2
Me![OrdenesDet].Form("T" & Tt).Visible = False
Me![OrdenesDet].Form("T" & Tt & "a").Visible = False
Me![OrdenesDet].Form("TA" & Tt).Requery
Next i
End If


If Len([Tallas]) > 4 Then
Dim j, Tam1, Tt1, Di As Integer
 Di = Len([Tallas])
 For j = 4 To Di Step 2
 Tt1 = j / 2
 Me![OrdenesDet].Form("T" & Tt1).Visible = True
 Me![OrdenesDet].Form("T" & Tt1 & "a").Visible = True
 Me![OrdenesDet].Form("TA" & Tt1).Requery
Next j
End If
End Sub

Private Sub CopiarDetallesOrd()
On Error GoTo errores
Dim a, i As Integer
Dim Dbs As Database
Dim RstDet As Recordset

a = IdOrdenesAnt
Set Dbs = CurrentDb
Set RstDet = Dbs.OpenRecordset("SELECT IdOrdenes,Color,T1,T2,T3,T4,T5,T6,T7,T8 FROM OrdenesDet WHERE IdOrdenes=" & a)

RstDet.MoveLast
RstDet.MoveFirst
Do While Not RstDet.EOF
Me![OrdenesDet].Form![Color] = RstDet!Color
 For i = 1 To 8
 Me![OrdenesDet].Form("T" & i) = RstDet("T" & i)
 Next
Me![OrdenesDet].Form.NuevoRecor
RstDet.MoveNext

Loop
RstDet.Close
Dbs.Close
Exit Sub

errores:
MsgBox Err.Description
End Sub


Private Sub CargarFotos()
On Error GoTo Errorprueba
Dim Nu As Integer
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Nu = 1
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Nu = 2
Me!Imagen2.Picture = DirFotos(Foto2)
Me!Imagen2.HyperlinkAddress = DirFotos(Foto2)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
Else
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
Case 2
Me!Imagen2.Picture = NoFoto
End Select
End If
Resume Next
End Sub
