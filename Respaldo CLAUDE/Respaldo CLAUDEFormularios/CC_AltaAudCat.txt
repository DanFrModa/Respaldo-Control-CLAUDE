Private Sub Agreg()
Dim Dbs As Database
Dim QueIdAud As Integer
QueIdAud = Forms![CC_AltaAuditorias]![IdCC_Auditorias]
Set Dbs = CurrentDb
Dbs.Execute "INSERT INTO CC_AuditoriasDet (IdCC_Auditorias,IdCC_Catalogo) VALUES (" & QueIdAud & "," & Me.IdCC_Catalogo & ")"
Dbs.Close
Forms![CC_AltaAuditorias]![CC_AltaAuditDet].Form.Requery
End Sub

Private Sub Copiar_Click()
On Error GoTo Erro
Dim Dbs As Database
Dim Rst As Recordset
Dim Dup As Boolean
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdCC_Catalogo, IdCC_Auditorias FROM CC_AuditoriasDet WHERE IdCC_Auditorias=" & Forms![CC_AltaAuditorias]![IdCC_Auditorias])

Rst.MoveLast
Rst.MoveFirst
  
 Do While Not Rst.EOF
 Debug.Print Rst!IdCC_Catalogo
        If Rst!IdCC_Catalogo = Me.IdCC_Catalogo Then
           MsgBox "Ya existe ese punto de revision en esta Auditoria", vbInformation, "YA EXISTE ESE PUNTO DE AUDITORIA"
           GoTo SiDup
        End If
            Rst.MoveNext
     Loop
Siguele:
Call Agreg

SiDup:

Rst.Close
Dbs.Close
Me.Requery

Exit Sub
Erro:
If Err.Number = 3021 Then
 GoTo Siguele
End If
End Sub
