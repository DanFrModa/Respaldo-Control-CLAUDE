Private Sub Transferir_Click()
If IsNull(AlmSal) = True Or IsNull(AlmEnt) = True Then
 MsgBox "Debes de seleccionar un almacen de salida y un almacen de entrada para poder hacer la transferencia entre almacenes", vbInformation, "FALTA DETERMINAR ALMACENES"
Else
DoCmd.OpenForm ("ITelas_TransferAlm")
Forms![ITelas_TransferAlm]![QueIdAlmSal] = AlmSal
Forms![ITelas_TransferAlm]![QueIdAlmEnt] = AlmEnt

Dim Dbs As Database
Dim Rst, Rst1 As Recordset

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Almacen FROM Almacenes WHERE IdAlmacenes=" & AlmSal)
Set Rst1 = Dbs.OpenRecordset("SELECT Almacen FROM Almacenes WHERE IdAlmacenes=" & AlmEnt)

Forms![ITelas_TransferAlm]![QueAlmSal] = Rst!Almacen
Forms![ITelas_TransferAlm]![QueAlmEnt] = Rst1!Almacen
Rst.Close
Rst1.Close
Dbs.Close
End If


End Sub
