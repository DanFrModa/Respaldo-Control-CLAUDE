Private Sub AplicarFiltro_Click()
If AplicarFiltro = -1 Then
Call FiltrarInfo
Else
DoCmd.ShowAllRecords
End If
End Sub

Private Sub Comando24_Click()
If IsNull(Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![Costo]) = False Then
  MsgBox "Ya esta relacionado este modelo con una orden de produccion. No es posible meter un costo manual o ponderado, mientras exista una relacion. Borra primero la relacion para poder meter este costo", vbInformation
Else
Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![CostoViejo] = CostoPonderado
DoCmd.Close
End If
End Sub

Private Sub ComoBuscar_AfterUpdate()
Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![QueNum] = ComoBuscar
If ComoBuscar = 3 Then
    [QueOrden].Visible = True
    [QueModelo].Visible = True
    [QueCliente].Visible = True
Else
    [QueOrden].Visible = False
    [QueModelo].Visible = False
    [QueCliente].Visible = False
End If

Call FiltrarInfo
End Sub

Private Sub FechaDe_AfterUpdate()
Call FiltrarInfo
End Sub

Private Sub FechaHasta_AfterUpdate()
Call FiltrarInfo
End Sub

Private Sub Form_Close()
Forms![EdoResultBuscar]![QueModeloEs] = "*"
Forms![EdoResultBuscar]![QueClienteEs] = "*"
End Sub

Private Sub Form_Open(Cancel As Integer)
QModeloC = Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![ModeloC]
QCantVendida = Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![CantVendida]
QPrecioVenta = Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![PrecioVenta]

FechaDe = Forms![EdoResultBuscar]![Fecha] - 365
FechaHasta = Forms![EdoResultBuscar]![Fecha]

Call FiltrarInfo
End Sub

Private Sub QueCliente_AfterUpdate()

End Sub

Private Sub QueModelo_AfterUpdate()

Else

End If

End Sub

errores:
  
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub

Private Sub Seleccionar_Click()
If (Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![CostoViejo]) <> 0 Then
MsgBox "No puedes relacionar una orden de produccion, por que existe un costo manual. Primero debes de porner en ceros el costo manual para continuar", vbInformation
Else
Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![IdCostoOrd] = IdCostoOrd
DoCmd.Close
End If
End Sub

Public Sub FiltrarInfo()
Dim LaFecha, LaFecha2 As Date

If IsNull(FechaDe) = True Then
LaFecha = #1/1/2003#
Else
LaFecha = Format(FechaDe, "mm/dd/yy")
End If

If IsNull(FechaHasta) = True Then
LaFecha2 = #1/1/2020#
Else
LaFecha2 = Format(FechaHasta, "mm/dd/yy")
End If


If Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![QueNum] = 1 Then
DoCmd.ApplyFilter , "[Fecha] >= (#" & LaFecha & "#) and [fecha] <= (#" & LaFecha2 & "#) and [modelo] like '" & QueModFiltro & "'"
ElseIf Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![QueNum] = 2 Then
DoCmd.ApplyFilter , "[Fecha] >= (#" & LaFecha & "#) and [fecha] <= (#" & LaFecha2 & "#) and [monarch] like '" & QueModFiltro & "'"
ElseIf Forms![EdoResultBuscar]![EdoResultBuscarSub].Form![QueNum] = 3 Then
DoCmd.ApplyFilter , "[Fecha] >= (#" & LaFecha & "#) and [fecha] <= (#" & LaFecha2 & "#)"
End If
End Sub
