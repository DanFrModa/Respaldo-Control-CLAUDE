Private Sub Form_Close()
On Error GoTo errores
Dim Rst As Recordset
Me![OrdDetEntM].Form.Recalc
Set Rst = CurrentDb.OpenRecordset("SELECT OrdenesDetEntM.IdEntregas, Sum([TC1]+[tc2]+[tc3]+[tc4]+[tc5]+[tc6]+[tc7]+[tc8]) AS Total FROM OrdenesDetEntM GROUP BY OrdenesDetEntM.IdEntregas HAVING (OrdenesDetEntM.IdEntregas)=" & IdEntregas)

If IsNull(Rst!Total) = True Or (Rst.EOF And Rst.BOF) Then
CurrentDb.Execute "UPDATE Entregas SET Cantidad=0 WHERE IdEntregas=" & IdEntregas
Else
CurrentDb.Execute "UPDATE Entregas SET Cantidad=" & Rst!Total & " WHERE IdEntregas=" & IdEntregas

End If
Rst.Close

errores:
If Err.Number = 3021 Then

End If

If EstaCargado("Proceso") = True Then

Forms![Proceso]![ProcesoEntrega].Form.RecordsetClone.FindFirst "[IdEntregas] = " & IdEntregas
   Forms![Proceso]![ProcesoEntrega].Form.Bookmark = Forms![Proceso]![ProcesoEntrega].Form.RecordsetClone.Bookmark
End If
End Sub
