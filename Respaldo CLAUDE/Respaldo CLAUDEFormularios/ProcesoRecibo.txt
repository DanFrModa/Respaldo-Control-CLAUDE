Option Compare Database
Option Explicit
Dim Ca As Integer

Private Sub Cant_Click()

If IsNull(Maquilero) = True Then
 MsgBox "Debes de registrar el maquilero al que se le va a recibir para poder continuar.", vbInformation
 
Else

DoCmd.OpenForm ("OrdDetRecM")
Forms![OrdDetRecM]![QueIdOrdenes] = IdOrdenes
Forms![OrdDetRecM]![QueIdRecibos] = IdRecibos
Forms![OrdDetRecM]![TallasLoc] = Forms![Proceso]![Tallas]
Forms![OrdDetRecM].BorrarAa
Forms![OrdDetRecM]![IdOrdenesDet].Requery

End If
End Sub

Private Sub Cantidad_AfterUpdate()
If Forms![Proceso]![TotRecibo] + Me!Cantidad - Ca > Forms![Proceso]![TotEntrega] Then
MsgBox "No puede ser mayor la cantidad recibida que la cantidad entregada!!!. Checa bien el dato."
End If
End Sub

Private Sub Cantidad_GotFocus()
Ca = Me!Cantidad
End Sub

Private Sub Cantidad_LostFocus()
If Me!Cantidad = 0 Then
Me!Cantidad = Forms![Proceso]![Cant]
End If
End Sub


Private Sub Comando23_Click()

If Inventariado = True Then
MsgBox "Este movimiento ya esta inventariado"
Else
  If MsgBox("Despues de inventariar este movimiento no podras modificar la informacion que hay en el. Estas seguro de querer inventariar el movimiento?", vbYesNo, "INVENTARIAR MOVIMIENTO") = vbYes Then

  DoCmd.OpenForm "IPT_VerifMod"

  Call MeterInventario

  End If
End If
End Sub

End Sub

Private Sub MeterInventario()
If IsNull(Fecha) = True Then
MsgBox "Para poder continuar es necesario que metas una fecha.", vbInformation, "FALTA FECHA"
Else
  
If IsNull(Cantidad) = True Then
MsgBox "Para poder agregar un modelo es necesario que metas la cantidad.", vbInformation, "FALTA INFORMACION"
Else

Dim Dbs As Database
Dim UltMov, QueModOrd, QueModAlm As Recordset
Dim UltIdMov, QueIdModOrd As Integer
Set Dbs = CurrentDb
  Dbs.Execute "INSERT INTO IPT_Movs (Fecha,IdIPT_TipoMov,EnSa,TieneModelos,IdMaquileros,IdIPT_Almacenes,IdUsuarios,IdRecibos) VALUES (#" & Fecha & "#,2,1,1,'" & Maquilero & "','" & TipoPrendas & "','" & IdUsuarioACT & "','" & IdRecibos & "')"

Set QueModOrd = Dbs.OpenRecordset("SELECT IdIPT_Modelos AS ModOrd FROM IPT_Modelos WHERE Idordenes=" & Forms![Proceso]![IdOrdenes])
QueIdModOrd = QueModOrd!modord

Set QueModAlm = Dbs.OpenRecordset("SELECT IdIPT_Mod_Alm AS Ult FROM IPT_Mod_Alm WHERE IdIPT_Modelos=" & QueIdModOrd & " AND IdIPT_Almacenes= " & TipoPrendas)


Set UltMov = Dbs.OpenRecordset("SELECT Max(IdIPT_Movs) AS Ult FROM IPT_Movs")
 UltIdMov = UltMov!ult
  Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & UltIdMov & "','" & QueModAlm!ult & "','" & Cantidad & "')"
  Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia + " & Cantidad & " WHERE IdIPT_Mod_Alm = " & QueModAlm!ult
  
  Inventariado = True
 


QueModOrd.Close
QueModAlm.Close
UltMov.Close


 Dbs.Close
End If
End If
End Sub
