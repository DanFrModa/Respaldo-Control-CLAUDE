Private Sub CargarFotos_Click()

 Dim Rst As Recordset
 
 Set Dbs = CurrentDb
 Set Rst = Dbs.OpenRecordset("SELECT Modelos.Foto1, Modelos.Foto2, Bordados.Foto, Ordenes.IdOrdenes FROM Bordados RIGHT JOIN ((Modelos INNER JOIN Ordenes ON Modelos.IdModelos = Ordenes.IdModelos) LEFT JOIN ModelosBor ON Modelos.IdModelos = ModelosBor.IdModelos) ON Bordados.IdBordados = ModelosBor.IdBordados WHERE (Ordenes.IdOrdenes)=" & QueIdOrdenes)
    Foto1 = Rst!Foto1
    Foto2 = Rst!Foto2
    FotoBord = Rst!Foto
Rst.Close
Dbs.Close
errores:
Call CargarFot
Else
MsgBox "No hay ninguna orden relacionada a ese modelo. Solo se podran ver las fotos con una orden relacionada.", vbInformation, "NO HAY ORDEN RELACIONADA"
End If
End Sub

Private Sub CargarFot()
On Error GoTo Errorprueba
Dim Nu As Integer
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen3.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Me!Imagen3.HyperlinkAddress = ""
Nu = 1
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Nu = 2
Me!Imagen2.Picture = DirFotos(Foto2)
Me!Imagen2.HyperlinkAddress = DirFotos(Foto2)
Nu = 3
Me!Imagen3.Picture = DirBordados(FotoBord)
Me!Imagen3.HyperlinkAddress = DirBordados(FotoBord)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
MsgBox Err.Description
Else
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
Case 2
Me!Imagen2.Picture = NoFoto
Case 3
Me!Imagen3.Picture = NoFoto
End Select
End If
Resume Next
End Sub
