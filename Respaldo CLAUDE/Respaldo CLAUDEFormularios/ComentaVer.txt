Private Sub Form_Open(Cancel As Integer)
On Error GoTo errores
    Me.RecordsetClone.FindFirst "[IdComentaord] = " & Forms![OrdenVer]![Coment]
    Me.Bookmark = Me.RecordsetClone.Bookmark
Exit Sub
errores:
If Err.Number = 3021 Then
MsgBox "No hay ningun comentario en esta orden"
DoCmd.Close
Else
MsgBox Err.Description
End If
End Sub

If Me.FilterOn = True Then
   Me.FilterOn = False
End If
    
    Me.RecordsetClone.FindFirst "[IdComentaord] = " & QueComen
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub
