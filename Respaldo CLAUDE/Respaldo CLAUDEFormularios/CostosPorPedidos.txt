Private Sub FiltroMeses_AfterUpdate()
If FiltroMeses = 13 Then
QueMes = "*"
Else
QueMes = FiltroMeses
End If
Me![CostosPorPedidosDet].Requery
End Sub

Private Sub Form_Open(Cancel As Integer)
If EstaCargado("pedidospormes") = True Then
QueAno = Forms![PedidosPorMes]![QueAno]
QueMes = Forms![PedidosPorMes]![QueMes]
FiltroMeses = Forms![PedidosPorMes]![FiltroMeses]
Me![CostosPorPedidosDet].Requery
End If

If EstaCargado("EdoResultPorMes") = True Then

  QueAno = Forms![EdoResultPorMes]![EdoResultPorMesSub].Form![QueAnoEs]
  QueMes = Forms![EdoResultPorMes]![EdoResultPorMesSub].Form![QueMesEs]
     If IsNull(Forms![EdoResultPorMes]![EdoResultPorMesSub].Form![FechaHasta]) = False Then
     FiltroMeses = Month(Forms![EdoResultPorMes]![EdoResultPorMesSub].Form![FechaHasta])
     End If
Me![CostosPorPedidosDet].Requery
End If

End Sub
