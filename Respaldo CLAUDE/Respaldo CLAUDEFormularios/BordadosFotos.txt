Private Sub Form_Current()
ListaModelos.Requery
On Error GoTo Errorprueba
Me!Imagen1.Picture = DirBordados(Foto)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto)

Errorprueba:
If Err.Number = 94 Then
Me!Imagen1.Picture = NoFoto
Me!Imagen1.HyperlinkAddress = ""
End If
End Sub

Private Sub QueBordadoLista_AfterUpdate()

End Sub

Private Sub QueEs_AfterUpdate()

Else
QueEsFijo = QueEs
End If

End Sub
