Private Sub AltaMod_Click()
Dim Mens As String
Dim UltIdAlm As Integer
Mens = "Estas Seguro que quieres dar de Alta un nuevo Almacen de Prenda Terminada con el Nombre " & QueAlmacen & ",? (Una vez creado, es imposible borrarlo)"
If MsgBox(Mens, vbYesNo, "Alta de Almacen") = vbYes Then

Me.RecordsetClone.FindFirst "[Almacen] = '" & Me![QueAlmacen] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
   If Me!QueAlmacen = Me!Almacen Then
   MsgBox "Ese nombre de Almacen ya esta dado de alta. No es posible dar de alta dos Almacenes con el mismo nombre", vbInformation, "ALMACEN EXISTENTE"
     QueAlmacen.SetFocus
     GoTo ByeBye
   End If


Dim Dbs As Database
Dim Model, UltAlm As Recordset

Set Dbs = CurrentDb
Dbs.Execute "INSERT INTO IPT_Almacenes (Almacen) VALUES ('" & QueAlmacen & "')"

Set Model = Dbs.OpenRecordset("SELECT IdIPT_Modelos FROM IPT_Modelos")
Set UltAlm = Dbs.OpenRecordset("SELECT Last(IdIPT_Almacenes) AS Ult FROM IPT_Almacenes")
 UltIdAlm = UltAlm!ult
With Model
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            IdIPT_Modelos = ![IdIPT_Modelos]
            IdIPT_Almacenes = UltIdAlm
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
MsgBox "Se dio de alta el nuevo Almacen con exito.", vbInformation, "LISTO"
DoCmd.Close

End If
ByeBye:

End Sub

DoCmd.Close
