Private Function ContarReg() As Integer
Dim Dbs As Database
Dim pp As Recordset

Set Dbs = CurrentDb
Set pp = Dbs.OpenRecordset("Select idordenes from  rc where idordenes= " & Forms![RC_Programacion]![IdOrdenes])
pp.MoveLast

ContarReg = pp.RecordCount

pp.Close
Dbs.Close
End Function

Private Sub VerifAntecesor()
If AntecesorRC <> 0 Then
UnoMas:
            Me.RecordsetClone.FindFirst "[numprocesoRC] = " & Me!AntecesorRC
            If Me.RecordsetClone.NoMatch Then
              Me!AntecesorRC = Me!AntecesorRC - 1
              GoTo UnoMas
            End If
End If
End Sub

Private Function FactCant(TiempoCc As Integer) As Integer
Dim Dbs As Database
Dim Cc As Recordset
Dim Cantid As Long
Dim TiempoSim As Single

Cantid = Forms![RC_Programacion]![Cantidad]
Set Dbs = CurrentDb
Set Cc = Dbs.OpenRecordset("Select DeCant, ACant, FactorCant from CP_Cant")
With Cc
 .MoveLast
 .MoveFirst
                
Do While Not .EOF
  
   If Cantid >= !decant And Cantid <= !acant Then
     TiempoSim = ((TiempoCc) * (!factorcant)) + TiempoColchonCostura()
     FactCant = CInt(TiempoSim)
      If FactCant = 0 Then
       FactCant = 1
      End If
       
     GoTo ByeBye
   End If
   
            .MoveNext

       Loop

End With
MsgBox "No existe rango de cantidad definido, por lo tanto no se aplico el factor", vbInformation
FactCant = TiempoCc

ByeBye:
Cc.Close
Dbs.Close

End Function

Private Sub HacerProgramacion()
If IsNull(Forms![RC_Programacion]![FechaProg]) = False Then
MsgBox "Ya esta programada esta orden de produccion!!!", vbInformation, "YA ESTA PROGRAMADA"
Exit Sub
ElseIf IsNull(Forms![RC_Programacion]![Clave]) = True Then
MsgBox "No esta definido el tipo de producto de esta orden y por lo tanto no se puede programar.", vbInformation, "FALTA TIPO DE PRODUCTO"
Exit Sub
ElseIf IsNull(Forms![RC_Programacion]![FechaEntregaRC]) = True Then
MsgBox "Tienes que definir una Fecha de entrega de la RC, sin esta fecha no se puede programar.", vbInformation, "FALTA FECHA DE LA RC"
Exit Sub
ElseIf IsNull(Forms![RC_Programacion]![TipoTela]) = True Then
MsgBox "Tienes que definir un Tipo de Tela, sin esa informacion no se puede programar.", vbInformation, "FALTA TIPO DE TELA"
Exit Sub
ElseIf IsNull(Forms![RC_Programacion]![Aplic]) = True Then
MsgBox "Tienes que definir un Tipo de Aplicacion, sin esa informacion no se puede programar.", vbInformation, "FALTA APLICACION"
Exit Sub


Else

Me.AllowAdditions = True
Dim Dbs As Database
Dim Prog As Recordset

Set Dbs = CurrentDb
Set Prog = Dbs.OpenRecordset("SELECT CP_Tiempos.IdCP_Tiempos, CP_Tiempos.IdCP_Procesos, CP_Tiempos.IdCP_Articulos, CP_Tiempos.Antecesor, CP_Procesos.NumProceso,CP_Procesos.TipoProceso, CP_Procesos.NoLlevaProceso, CP_Tiempos.Tiempo FROM CP_Procesos INNER JOIN (CP_Articulos INNER JOIN CP_Tiempos ON CP_Articulos.IdCP_Articulos = CP_Tiempos.IdCP_Articulos) ON CP_Procesos.IdCP_Procesos = CP_Tiempos.IdCP_Procesos WHERE CP_Tiempos.IdCP_Articulos = " & QueArt)
        
With Prog
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             
               If Forms![RC_Programacion]![DiasAp] = 0 Then
                 If !NoLlevaProceso = True Then
                   GoTo AlSiguiente
                 End If
               End If
             
             DoCmd.GoToRecord , , acNewRec
         
            Me![IdCP_Tiempos] = !IdCP_Tiempos
            Me![IdCP_Procesos] = !IdCP_Procesos
            Me![NumProcesoRC] = !NumProceso
            Me![AntecesorRC] = !Antecesor
         
         If IsNull(!TipoProceso) = True Then
            Me![TiempoRC] = !Tiempo
         
         
         ElseIf !TipoProceso = "T" Then
                If IsNull(Forms![RC_Programacion]![TipoTela]) = False Then
                    Me![TiempoRC] = TelasDias()
                Else
                    MsgBox "Se esta metiendo la programacion SIN un tipo de telas especifico. Se va a tomar el tiempo predeterminado para este tipo de producto", vbInformation, "NO ESTA ESPECIFICADO EL TIPO DE TELA"
                    Me![TiempoRC] = !Tiempo
                End If
                
                
         ElseIf !TipoProceso = "RP" Then
            
                If IsNull(Forms![RC_Programacion]![IdRC_Aplicaciones]) = False Then
                    Me![TiempoRC] = FactCantAp(AplicDias())
                Else
                    MsgBox "Se esta metiendo la programacion SIN un Tipo de Aplicacion especifico. Se va a tomar el tiempo predeterminado para este tipo de producto", vbInformation, "NO ESTA ESPECIFICADO LAS APLICACIONES"
                    Me![TiempoRC] = !Tiempo
                End If
                
         ElseIf !TipoProceso = "EC" Then
                 If (Forms![RC_Programacion]![DiasAp] = 0) Then
                 
                    Me![AntecesorRC] = 22
                    Me![TiempoRC] = !Tiempo
                 Else
                    Me![AntecesorRC] = !Antecesor
                    Me![TiempoRC] = !Tiempo
                 End If
                 
         
         ElseIf !TipoProceso = "C" Then
            Me![TiempoRC] = FactCant(!Tiempo)
         
         Else
            Me![TiempoRC] = !Tiempo
         End If
            
AlSiguiente:
            .MoveNext

        Loop


        .Close
    End With
    

Dbs.Close
Me.AllowAdditions = False
 Forms![RC_Programacion]![FechaProg] = Now()
Me.Requery
Forms![RC_Programacion]![QueOrden].Requery
Call MeterDiasAcumulados

End If
End Sub

Private Sub EstablecerLasFechas()
Dim totalDias As Integer
Dim Aa As Boolean
Aa = True
DoCmd.GoToRecord , , acLast
totalDias = Acumulado
Dim FechaInicio, QueFecha, QueFechaAnt As Date

Dim i, numReg, QueReg, DiasDeMas, DiasDeMas2 As Integer
FechaInicio = FechaHabiles(QueFechaRC, totalDias)
OtraVez:
            DoCmd.GoToRecord , , acFirst
            FechaReal = Forms![RC_Programacion]![Fecha]
              If Aa = False Then
              FechaInicio = FechaInicio - 1
              Aa = True
              End If
            If WeekDay(FechaInicio) = vbSaturday Then
            FechaEst = FechaInicio - 1
            ElseIf WeekDay(FechaInicio) = vbSunday Then
            FechaEst = FechaInicio - 2
            Else
            FechaEst = FechaInicio
            End If
            
            
        For i = 1 To ContarReg() - 1
            DoCmd.GoToRecord , , acNext
            QueReg = Me.CurrentRecord
            
            Me.RecordsetClone.FindFirst "[numprocesorc] = " & Me!AntecesorRC
            Me.Bookmark = Me.RecordsetClone.Bookmark
            QueFechaAnt = FechaEst
            DoCmd.GoToRecord , , acGoTo, QueReg
            
            QueFecha = QueFechaAnt + TiempoRC
            DiasDeMas = CuantosSabYDom(QueFechaAnt, QueFechaAnt + TiempoRC)
            
            DiasDeMas2 = CuantosSabYDom(QueFechaAnt + TiempoRC, QueFechaAnt + TiempoRC + DiasDeMas)
            
            QueFecha = QueFecha + DiasDeMas + DiasDeMas2
            
            
            If WeekDay(QueFecha) = vbSaturday Then
             FechaEst = QueFecha + 2
            ElseIf WeekDay(QueFecha) = vbSunday Then
             FechaEst = QueFecha + 1
            Else
             FechaEst = QueFecha
            End If
            
               If TipoProceso = "C" Then
                 Forms![RC_Programacion]![FechaEntrega] = FechaEst
               End If
        Next
        
        
 If FechaEst > QueFechaRC Then
  Aa = False
  GoTo OtraVez
 End If
Forms![RC_Programacion]![QueOrden].Requery

DoCmd.GoToRecord , , acFirst
Forms![RC_Programacion]![RC_ProgramacionSub].Form![FechaReal].SetFocus
End Sub
