Private Sub Imprimir_Click()
Dim a, b, i As Integer
a = OrdenHasta - OrdenDe + 1
b = OrdenDe
If MsgBox("Estas mandando a imprimir " & a & " ordenes." & vbCr & "De la Orden Numero " & OrdenDe & ", hasta la " & OrdenHasta & vbCr & "Estas 100% Seguro???", vbYesNo, "VERSION NORMAL") = vbYes Then
 For i = a To 1 Step -1
 DoCmd.OpenForm "ordimp", acPreview, , "numero=" & b
 DoCmd.PrintOut acPages, 1, 1, acMedium
 DoCmd.Close acForm, "ordimp"
 b = b + 1
 
 Next
End If
End Sub
