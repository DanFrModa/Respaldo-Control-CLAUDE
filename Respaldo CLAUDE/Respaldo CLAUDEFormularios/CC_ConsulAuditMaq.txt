Private Sub Form_Current()
If Resultado = 1 Then
  Me.Queres.Caption = "APROBADO"
  Me.Queres.BackColor = 65280
ElseIf Resultado = 2 Then
  Me.Queres.Caption = "REPROBADO"
  Me.Queres.BackColor = 255
Else
  Me.Queres.Caption = " NO CALIFICADO"
  Me.Queres.BackColor = 16777215
End If

If Cancelada = True Then
  Cancel.Visible = True
Else
  Cancel.Visible = False
End If
End Sub

Private Sub Form_Open(Cancel As Integer)
If PrP(12) = True Then
  Modificar.Visible = True
End If
End Sub

Private Sub QueAuditoria_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdCC_Auditorias] = " & Me![QueAuditoria]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub QueAuditoria_DblClick(Cancel As Integer)

    Dim stLinkCriteria As String
    stDocName = "CC_ModificarDatos"
    stLinkCriteria = "[IdCC_Auditorias]=" & Me![IdCC_Auditorias]

End If
End Sub
