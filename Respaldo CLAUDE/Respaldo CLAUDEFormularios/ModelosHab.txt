Private Sub Comando13_Click()
Dim Dbs As Database
Dim CopiaHab As Recordset
Dim strmensaje As String

Set Dbs = CurrentDb
Set CopiaHab = Dbs.OpenRecordset("SELECT Habilitacion.IdHabilitacion, Habilitacion.Favorito,Habilitacion.CantFav, Habilitacion.Clave,Habilitacion.Descripcion FROM Habilitacion WHERE (((Habilitacion.Favorito)=Yes))")
        
With CopiaHab
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            strmensaje = "Clave: " & !Clave & vbCr & "Descripcion: " & !Descripcion & vbCr & "Cantidad Preestablecida: " & !CantFav & vbCr & vbCr & _
                "(Habilitacion " & (.AbsolutePosition + 1) & _
                " de " & .RecordCount & ")"
            If MsgBox(strmensaje, vbOKCancel, "AGREGAR HABILITACION") = vbCancel _
                Then GoTo OtroMas
        
            Me![IdHabilitacion] = !IdHabilitacion
            Me!CantHab = !CantFav
           
OtroMas:
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
End Sub
