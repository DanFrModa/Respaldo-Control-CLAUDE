Private Sub BuscarNota_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdSalidas] = " & BuscarNota
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub Form_BeforeInsert(Cancel As Integer)
IdTela.Visible = True
Me![IdTela].Locked = False
Me!IdTela.Enabled = True
End Sub

Private Sub Form_Current()
Me![SalidasNotasSub].Form![IdTelasColAlm].Requery
Me![IdTela].Locked = False
Me!IdTela.Enabled = True


With Me![SalidasNotasSub].Form
If IsNull(![Cardigan1]) = True Then
![TelaEnt2].Visible = False
![ExTela2].Visible = False
Else
![TelaEnt2].Visible = True
![ExTela2].Visible = True
End If
End With

End Sub

Private Sub Form_Open(Cancel As Integer)
If NivelAct <= nivAdminDir Then
BuscarNota.Visible = True
End If
End Sub

Private Sub IdTela_AfterUpdate()
Me![SalidasNotasSub].Form![IdTelasColAlm].Requery
With Me![SalidasNotasSub].Form
If IsNull(![Cardigan1]) = True Then
![TelaEnt2].Visible = False
![ExTela2].Visible = False
Else
![TelaEnt2].Visible = True
![ExTela2].Visible = True
End If
End With
End Sub
