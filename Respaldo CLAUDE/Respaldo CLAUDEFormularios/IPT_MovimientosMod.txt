Option Compare Database

Private Sub CambiarCant_Click()
If MsgBox("Estas seguro de querer cambiar la cantidad del movimiento? Esto va  a afectar al movimiento y a la existencia del modelo y no se puede deshacer?", vbYesNo, "CAMBIO DE CANTIDAD") = vbYes Then
 If QueEnSa = 1 Then
  Existencia = Existencia - CantMov + QueCant
  CantMov = QueCant
  MsgBox "La cantidad fue cambiada con exito.", vbInformation, "CAMBIO DE CANTIDAD"
 ElseIf QueEnSa = 2 Then
   If ((Existencia + CantMov - QueCant) < 0) And (PrP(30) = False) Then
     MsgBox "No puedes dejar en ceros la existencia!. Modifica la cantidad.", vbInformation, "NO PUEDES DEJAR EN CERO LA EXISTECIA"
   Else
     Existencia = Existencia + CantMov - QueCant
     CantMov = QueCant
     MsgBox "La cantidad fue cambiada con exito.", vbInformation, "CAMBIO DE CANTIDAD"
   End If
 End If
QueCant = ""
End If
End Sub
