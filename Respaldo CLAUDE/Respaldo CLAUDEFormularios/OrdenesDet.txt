Private Sub Copiar_Click()
On Error GoTo errores
Dim i, TtAct As Integer
For i = 1 To 8
 DoCmd.GoToRecord , , acPrevious
  TtAct = Me("T" & i)
 DoCmd.GoToRecord , , acNext
 Me("T" & i) = TtAct
Next i
Exit Sub

errores:
MsgBox Err.Description
End Sub



Private Sub CopiarDetalles_Click()
On Error GoTo errores
If IsNull(Forms![Ordenes]![QueOrdCop]) = True Then
MsgBox "Debes de seleccionar una orden de la cual se van a copiar los colores"
Else

Dim a, i As Integer
Dim Dbs As Database
Dim RstDet As Recordset

a = Forms![Ordenes]![QueOrdCop]
Set Dbs = CurrentDb
Set RstDet = Dbs.OpenRecordset("SELECT IdOrdenes,Color,T1,T2,T3,T4,T5,T6,T7,T8 FROM OrdenesDet WHERE IdOrdenes=" & a)

RstDet.MoveLast
RstDet.MoveFirst
Do While Not RstDet.EOF
Me![Color] = RstDet!Color
 For i = 1 To 8
 Me("T" & i) = RstDet("T" & i)
 Next
DoCmd.GoToRecord , , acNewRec
RstDet.MoveNext

Loop
RstDet.Close
Dbs.Close
Forms![Ordenes]![FechaDet] = Now()
Exit Sub
End If
errores:
MsgBox Err.Description
End Sub

Private Sub Form_BeforeInsert(Cancel As Integer)
If IsNull(Forms![Ordenes]![FechaDet]) = True Then
 Forms![Ordenes]![FechaDet] = Now()
End If
End Sub

Private Sub Form_GotFocus()
If Len(Forms![Ordenes]![Tallas]) < 16 Then
Dim i, Tam, Tt As Integer
Tam = Len(Forms![Ordenes]![Tallas]) + 2
For i = 16 To Tam Step -2
Tt = i / 2
Me("T" & Tt).Visible = False
Me("T" & Tt & "a").Visible = False
Next i
End If

If Len(Forms![Ordenes]![Tallas]) > 4 Then
Dim j, Tam1, Tt1, Di As Integer
 Di = Len(Forms![Ordenes]![Tallas])
 For j = 4 To Di Step 2
 Tt1 = j / 2
 Me("T" & Tt1).Visible = True
 Me("T" & Tt1 & "a").Visible = True
Next j
End If
End Sub

Private Sub T1_LostFocus()

If Forms![Ordenes]![AutoLL] = 2 Then
  If T1a > 0 Then
    T2 = T2a / T1a * T1
    T3 = T3a / T1a * T1
    T4 = T4a / T1a * T1
    T5 = T5a / T1a * T1
    T6 = T6a / T1a * T1
    T7 = T7a / T1a * T1
    T8 = T8a / T1a * T1

  Else
  MsgBox "Debes de cambiar el valor de los campos de PROPORCION!!!"
  End If


End If
End Sub
