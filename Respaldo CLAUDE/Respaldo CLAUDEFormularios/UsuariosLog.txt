Private Sub Abiertas_AfterUpdate()
If Abiertas = True Then
Me.RecordSource = "SELECT [nombre]+[apellidos] AS Usua, FechaEntrada, FechaSalida, UsuariosLog.IdUsuarios, IdUsuariosLog, [fechaentrada]-[fechasalida] AS TotTiempo FROM UsuariosLog INNER JOIN Usuarios ON UsuariosLog.IdUsuarios = Usuarios.IdUsuarios GROUP BY [nombre]+[apellidos], FechaEntrada, FechaSalida, UsuariosLog.IdUsuarios, IdUsuariosLog HAVING (((UsuariosLog.FechaSalida) Is Null)) ORDER BY UsuariosLog.FechaEntrada DESC"
QueUsuario = ""
Else
Me.RecordSource = "SELECT [nombre]+[apellidos] AS Usua, UsuariosLog.FechaEntrada, UsuariosLog.FechaSalida, UsuariosLog.IdUsuarios, UsuariosLog.IdUsuariosLog, [fechaentrada]-[fechasalida] AS TotTiempo FROM UsuariosLog INNER JOIN Usuarios ON UsuariosLog.IdUsuarios = Usuarios.IdUsuarios GROUP BY [nombre]+[apellidos], UsuariosLog.FechaEntrada, UsuariosLog.FechaSalida, UsuariosLog.IdUsuarios, UsuariosLog.IdUsuariosLog HAVING (((UsuariosLog.IdUsuarios) Like [Formularios]![UsuariosLog]![queusuarioes])) ORDER BY UsuariosLog.FechaEntrada DESC"
End If
End Sub

Private Sub CerrarSesion_Click()
If IsNull(FechaSalida) = True Then
Dim Dbs As Database
Set Dbs = CurrentDb
Dbs.Execute ("UPDATE UsuariosLog SET FechaSalida = '" & Now & "' WHERE IdUsuariosLog = " & IdUsuariosLog)
Dbs.Execute ("UPDATE Usuarios SET EstaLogiado = (EstaLogiado - 1) WHERE IdUsuarios = " & IdUsuarios)
Dbs.Close
Me.Requery
Else
MsgBox "Ya esta cerrada esa sesion", vbInformation
End If
End Sub
