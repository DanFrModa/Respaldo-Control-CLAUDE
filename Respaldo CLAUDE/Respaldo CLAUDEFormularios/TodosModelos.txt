Private Sub QueGenero_AfterUpdate()

QueGeneroOculto = "*"
Else
QueGeneroOculto = QueGenero
End If

End Sub

    Dim stLinkCriteria As String

    DoCmd.OpenForm stDocName, , , stLinkCriteria

End Sub
