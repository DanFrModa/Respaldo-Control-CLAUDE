Private Sub CualAlmacen_AfterUpdate()
   Me.RecordsetClone.FindFirst "[IdAlmacenes] = " & Me![CualAlmacen]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub CualAlmacen_DblClick(Cancel As Integer)
If QueTipoSalida = 1 Then
DoCmd.OpenForm "SalidasOrd", , , , acFormAdd
Forms![SalidasOrd]![QueAlmacen] = Almacen
Forms![SalidasOrd]![QueIdAlmacenes] = IdAlmacenes
DoCmd.Close acForm, "QueAlmacenSalida"
Else
  If NivelAct <= nivAdminDir Then
  DoCmd.OpenForm "SalidasNotas"
  Else
  DoCmd.OpenForm "SalidasNotas", , , , acFormAdd
  End If
Forms![SalidasNotas]![QueAlmacen] = Almacen
Forms![SalidasNotas]![QueIdAlmacenes] = IdAlmacenes
DoCmd.Close acForm, "QueAlmacenSalida"
End If
End Sub
