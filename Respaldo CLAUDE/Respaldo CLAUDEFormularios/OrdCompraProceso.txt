Option Compare Database
Option Explicit

Private Sub Autoriz_Click()
If IdUsuarioACT <= nivDirectivo Then
If MsgBox("Estas seguro de querer autorizar la orden de compra numero " & NumCompra & " de " & Proveedor & "?", vbYesNo, "AUTORIZAR ORDEN") = vbYes Then
Salir.SetFocus
NoAutEtiqueta.Visible = False
AutEtiqueta.Visible = True
Autoriz.Visible = False
IdUsuAutorizado = IdUsuarioACT
Autorizado = True
FechaAutorizado = Now()
DoCmd.Close
End If
Else
MsgBox "No tienes permiso para autorizar ordenes de produccion"
End If
End Sub

Private Sub CancelarOrden_Click()
If MsgBox("Estas seguro de querer cancelar la orden de compra numero " & NumCompra & " de " & Proveedor & "?", vbYesNo, "CANCELAR ORDEN") = vbYes Then
Estatus = 2
IdUsuCancelado = IdUsuarioACT
Call PintarEstatus
CanceladoMotivo.SetFocus
End If
End Sub

Private Sub DAutoriz_Click()
If IdUsuarioACT <= nivAdminDir Then
If MsgBox("Estas seguro de querer DES-AUTORIZAR la orden de compra numero " & NumCompra & " de " & Proveedor & "?. Esta Accion borrara el registo de quien y cuando se autorizo originalmente esta orden.", vbYesNo, "AUTORIZAR ORDEN") = vbYes Then
Salir.SetFocus
NoAutEtiqueta.Visible = True
AutEtiqueta.Visible = False
Autoriz.Visible = True
IdUsuAutorizado = 0
Autorizado = False
FechaAutorizado = ""
DAutoriz.Visible = False
End If
Else
MsgBox "No tienes permiso para DES-AUTORIZAR ordenes de produccion"
End If
End Sub

Private Sub Form_Close()
If EstaCargado("ordcompraver") = True Then
Forms![OrdCompraVer].Requery
Forms![OrdCompraVer]![QueOrden].Requery
Forms![OrdCompraVer]![QueOrden] = Forms![OrdCompraVer]!QueOrdenEs

 Forms![OrdCompraVer].RecordsetClone.FindFirst "[IdOrdCompra] = " & Forms![OrdCompraVer]!QueOrdenEs
    Forms![OrdCompraVer].Bookmark = Forms![OrdCompraVer].RecordsetClone.Bookmark
     


End If
End Sub

Private Sub Form_Current()
Call PintarEstatus

If Autorizado = True Then
Autoriz.Visible = False
NoAutEtiqueta.Visible = False
AutEtiqueta.Visible = True
DAutoriz.Visible = True
Else
Autoriz.Visible = True
NoAutEtiqueta.Visible = True
AutEtiqueta.Visible = False
DAutoriz.Visible = False
End If

If IsNull(Excel) = True Then
Excel.Visible = False
End If
End Sub

Private Sub Form_Open(Cancel As Integer)
If PrP(8) = True Then
Autorizar.Visible = True
Autorizar.SetFocus
End If
End Sub

Private Sub ModificarDes_Click()
If MsgBox("Estas seguro que quieres modificar la hoja de excel que muestra el desglose de esta orden?", vbYesNo, "Modificar Desglose") = vbYes Then
DoCmd.Close acForm, "Ordcompraver"
DoCmd.OpenForm "OrdCompraExcel", acNormal, , "idordcompra =" & IdOrdCompra
Forms![OrdCompraExcel]![NumeroCom] = NumCompra
DoCmd.Close acForm, "Ordcompraproceso"
End If
End Sub

Private Sub OrdenRecibida_Click()
If Autorizado = True Then
 If MsgBox("Estas seguro de querer poner como RECIBIDO la orden de compra numero " & NumCompra & " de " & Proveedor & "?", vbYesNo, "CANCELAR ORDEN") = vbYes Then
 Estatus = 3
 Call PintarEstatus
 FechaRecibido.SetFocus
 End If
Else
 MsgBox "Para marcar la orden como 'Recibida', tiene que estar primero autorizada!.", vbInformation
End If
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub
Private Sub PintarEstatus()
If Estatus = 1 Then
CanceladaEs.BackColor = 16777215
CanceladaEs.Caption = "VIGENTE"
CanceladoMotivo.Visible = False
FechaRecibido.Visible = False
CancelarOrden.Visible = True
OrdenRecibida.Visible = True

ElseIf Estatus = 2 Then
CanceladaEs.BackColor = 255
CanceladaEs.Caption = "CANCELADA"
CanceladoMotivo.Visible = True
FechaRecibido.Visible = False
Salir.SetFocus
CancelarOrden.Visible = False
OrdenRecibida.Visible = False

ElseIf Estatus = 3 Then
