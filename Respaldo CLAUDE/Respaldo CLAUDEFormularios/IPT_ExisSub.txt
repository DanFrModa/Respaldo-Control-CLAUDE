Private Sub AbrirMovs()
Dim Dbs As Database
Dim Rst As Recordset
Dim CAlm, CMod As Integer
 If Forms![IPT_Exis]![QueAlmacenEs] = "*" Then
  CAlm = AlmDefPT
 Else
  CAlm = Forms![IPT_Exis]![QueAlmacenEs]
 End If

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdIPT_Mod_Alm FROM IPT_Mod_Alm WHERE (IdIPT_Modelos=" & IdIPT_Modelos & ") AND IdIPT_Almacenes=" & CAlm)
CMod = Rst!IdIPT_Mod_Alm
Rst.Close
Dbs.Close

DoCmd.OpenForm "IPT_MovsSaldo"
Forms![IPT_MovsSaldo]![QueAlm] = CAlm
Forms![IPT_MovsSaldo]![QueMod].Requery
Forms![IPT_MovsSaldo]![QueMod] = CMod

Forms![IPT_MovsSaldo].Requery
Forms![IPT_MovsSaldo]![ExisAlm].Requery
End Sub
