Private Sub Form_Current()
OrdenesSalidas.Requery

On Error GoTo Errorprueba

Dim Nu As Integer
Me!Imagen1.Picture = ""
Me!Imagen1.HyperlinkAddress = ""

Nu = 1
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
Else
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
End Select
End If
Resume Next
End Sub

QueOrden.RowSource = "SELECT Ordenes.IdOrdenes, Ordenes.Numero, Modelos.Modelo, Corte.Cantidad, Maquileros.Corto, Clientes.Cliente FROM Maquileros RIGHT JOIN (((((Ordenes LEFT JOIN Salidas ON Ordenes.IdOrdenes = Salidas.IdOrdenes) INNER JOIN Modelos ON Ordenes.IdModelos = Modelos.IdModelos) INNER JOIN Corte ON Ordenes.IdOrdenes = Corte.IdOrdenes) INNER JOIN Clientes ON Ordenes.IdClientes = Clientes.IdClientes) LEFT JOIN Entregas ON Ordenes.IdOrdenes = Entregas.IdOrdenes) ON Maquileros.IdMaquileros = Entregas.IdMaquileros GROUP BY Ordenes.IdOrdenes, Ordenes.Numero, Modelos.Modelo, Corte.Cantidad, Maquileros.Corto, Clientes.Cliente HAVING (((Ordenes.Numero)<>0)) ORDER BY Ordenes.Numero DESC"
