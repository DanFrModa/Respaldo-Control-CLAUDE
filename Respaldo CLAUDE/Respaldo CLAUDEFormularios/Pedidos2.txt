Private Sub CancelarPed_Click()
If MsgBox("Estas seguro de querer cancelar el pedido numero " & NumeroPed & "?", vbYesNo, "CANCELAR PEDIDO") = vbYes Then

End If
End Sub

Private Sub CopiarPed_Click()
If (MsgBox("Estas seguro que quieres copiar el pedido " & NumeroPed & " en uno nuevo?", vbYesNo, "COPIAR PEDIDO")) = vbYes Then
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
IdPedidosAnt = IdPedidos
Set Rst = Dbs.OpenRecordset("SELECT IdPedidos, IdClientes, FechaPedido FROM Pedidos WHERE IdPedidos=" & IdPedidos)
DoCmd.GoToRecord , , acNewRec
Me!IdClientes = Rst!IdClientes
Me!FechaPedido = Rst!FechaPedido
Me!NumeroPed = 0
Rst.Close

Dbs.Close
QuePedidoLis.Requery
Me![PedidosDet].Form![CopiarMod].Visible = True
End If
End Sub

Private Sub Form_Close()

Private Sub Form_Current()
If PedCancelado = True Then
Cancelado.Visible = True
CancelarPed.Visible = False
Else
Cancelado.Visible = False
CancelarPed.Visible = True
End If
End Sub

Private Sub Form_Open(Cancel As Integer)
If EstaCargado("PedidosPorMes") = True Then
Me.DataEntry = False
Me.Modificar = True
Me.QuePedidoLis = Forms![PedidosPorMes]![IdPedidos]
   Me.RecordsetClone.FindFirst "[IdPedidos] = " & Forms![PedidosPorMes]![IdPedidos]
    Me.Bookmark = Me.RecordsetClone.Bookmark
Else
Me.QuePedido.Visible = False
Me.QueCliente.Visible = False
Me.QuePedidoLis.Visible = False
End If
End Sub

Private Sub Modificar_AfterUpdate()
If Modificar = True Then
QuePedido.Visible = True
QueCliente.Visible = True
QuePedidoLis.Visible = True
Me.DataEntry = False
Else
QuePedido.Visible = False
QueCliente.Visible = False
QuePedidoLis.Visible = False
Me.DataEntry = True
End If
End Sub

Private Sub QueCliente_AfterUpdate()
If IsNull(QueCliente) = False Then
QuePedido = Null
QueClienteEs = QueCliente
QuePedidoLis.Requery
Else
QueClienteEs = "*"
QuePedidoLis.Requery
End If
End Sub

Private Sub QuePedido_AfterUpdate()
If IsNull(QuePedido) = False Then
QueCliente = Null
QueClienteEs = "*"
QuePedidoLis.Requery
QuePedidoLis = QuePedido
    Me.RecordsetClone.FindFirst "[IdPedidos] = " & Me![QuePedido]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End If
End Sub

Private Sub AumentarNumPed()
On Error GoTo errores
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Max(Pedidos.NumeroPed) AS UltNum FROM Pedidos")
Me!NumeroPed = Rst!ultnum + 1
Rst.Close
Dbs.Close
Exit Sub
errores:
MsgBox Err.Description
End Sub
