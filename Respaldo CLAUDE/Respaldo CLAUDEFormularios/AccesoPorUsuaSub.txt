Option Compare Database

Private Sub Copiar_Click()
On Error GoTo Errures
Dim Dbs As Database
Dim Cop As Recordset

Set Dbs = CurrentDb

Set Cop = Dbs.OpenRecordset("SELECT UsuAccesos.IdAccesos, UsuAccesos.[Activado], UsuAccesos.IdUsuarios FROM UsuAccesos WHERE IdUsuarios=" & QueUsuaCop)
        Cop.MoveLast
        Cop.MoveFirst
 DoCmd.GoToRecord , , acFirst
        Do While Not Cop.EOF
        
          Me![Activado] = Cop![Activado]
            
            Cop.MoveNext
            DoCmd.GoToRecord , , acNext
        Loop
Errures:

        Cop.Close
    
Dbs.Close


End Sub
