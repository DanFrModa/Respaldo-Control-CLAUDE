Private Sub IdTelasColores_AfterUpdate()
Dim Dbs As Database
Dim rstEnt, rstSal As Recordset


Set Dbs = CurrentDb

Set rstSal = Dbs.OpenRecordset("SELECT IdTelasColAlm,ExTela1,ExTela2 FROM TelasColAlm Where IdTelasColores=" & IdTelasColores & " And IdAlmacenes=" & Forms![ITelas_TransferAlm]![QueIdAlmSal])
Me![ExTelSal1] = rstSal!ExTela1
Me![ExTelSal2] = rstSal!ExTela2
IdTelasColAlmSalT = rstSal!IdTelasColAlm

Set rstEnt = Dbs.OpenRecordset("SELECT IdTelasColAlm,ExTela1,ExTela2 FROM TelasColAlm Where IdTelasColores=" & IdTelasColores & " And IdAlmacenes=" & Forms![ITelas_TransferAlm]![QueIdAlmEnt])
Me![ExTelEnt1] = rstEnt!ExTela1
Me![ExTelEnt2] = rstEnt!ExTela2
IdTelasColAlmEntT = rstEnt!IdTelasColAlm

rstSal.Close
rstEnt.Close
Dbs.Close

End Sub

Private Sub Transferir_Click()

If IsNull(Forms![ITelas_TransferAlm]![QueFecha]) = True Then
MsgBox "Para poder continuar es necesario que metas una fecha.", vbInformation, "FALTA FECHA"
Forms![ITelas_TransferAlm]![QueFecha].SetFocus

ElseIf IsNull(Forms![ITelas_TransferAlm]![QueTela]) = True Then
 MsgBox "Para poder continuar es necesario que selecciones una Tela.", vbInformation, "FALTA TELA"
Forms![ITelas_TransferAlm]![QueTela].SetFocus
Else
Me.Requery
Dim Dbs As Database
Dim rstTemp, rstUltS, rstUltE As Recordset

Set Dbs = CurrentDb
Set rstTemp = Dbs.OpenRecordset("SELECT IdITelas_Temp, IdTelasColores, CantTemp1, CantTemp2, IdTelasColAlmEntT, IdTelasColAlmSalT, ExTelSal1, ExTelSal2, ExTelEnt1, ExTelEnt2 FROM ITelas_Temp")
        
With rstTemp

  If .EOF = True Then
  MsgBox "Debes de meter por lo menos un color y la cantidad a trasferir.", vbInformation, "FALTA INFORMACION"
  
  
  ElseIf MsgBox("Estas seguro de querer hacer la transferencia de estas telas entre estos almacenes? Una vez realizado, es imposible deshacerlo!", vbYesNo) = vbYes Then



' Se mete el moviemiento de entrada/salida en la tabla padre.
      Dbs.Execute "INSERT INTO Salidas (Fecha,IdTela,Referencia) VALUES ('" & Forms![ITelas_TransferAlm]![QueFecha] & "','" & Forms![ITelas_TransferAlm]![QueTela] & "','" & Forms![ITelas_TransferAlm]![QueAlmEnt] & "')"
      Dbs.Execute "INSERT INTO Entradas (Fecha,IdTela,Factura,Referencia) VALUES ('" & Forms![ITelas_TransferAlm]![QueFecha] & "','" & Forms![ITelas_TransferAlm]![QueTela] & "','Transferencia','" & Forms![ITelas_TransferAlm]![QueAlmSal] & "')"
      
' Se saca la ultima salida/entrada para relacionar la tabla __Det
      Set rstUltS = Dbs.OpenRecordset("SELECT Max(Salidas.IdSalidas) AS UltIdSal FROM Salidas")
      Set rstUltE = Dbs.OpenRecordset("SELECT Max(Entradas.IdEntradas) AS UltIdEnt FROM Entradas")
      
        .MoveLast
        .MoveFirst
        
' Para cada registro se valida que no este en ceros para y que no deje el almacen en negativo:
        Do While Not .EOF
            If (IsNull(!CantTemp1) = True Or !CantTemp1 = 0) And (IsNull(!CantTemp2) = True Or !CantTemp2 = 0) Then
             MsgBox "Los colores que esten en ceros no se agregaran a esta transferencia"
            GoTo noagregar
            End If
            
            If (!ExTelSal1 - !CantTemp1 < 0) Or (!ExTelSal2 - !CantTemp2 < 0) Then
             MsgBox "No se puede sacar mas tela de la que hay en el almacen. No se registro el moviemiento de los colores que iban a dejar el almacen de salida con una cantidad en negativo."
            Dbs.Execute ("DELETE * FROM ITelas_Temp WHERE IdITelas_Temp=" & !IdITelas_Temp)
            
            GoTo noagregar
            End If
            
' Meterlo en ls tablas ___Det
     Dbs.Execute "INSERT INTO SalidasDet (IdSalidas,IdTelasColAlm,TelaSal1,TelaSal2) VALUES ('" & rstUltS!UltIdSal & "','" & !IdTelasColAlmSalT & "','" & !CantTemp1 & "','" & !CantTemp2 & "')"
     Dbs.Execute "INSERT INTO EntradasDet (IdEntradas,IdTelasColAlm,TelaEnt1,TelaEnt2) VALUES ('" & rstUltE!ultident & "','" & !IdTelasColAlmEntT & "','" & !CantTemp1 & "','" & !CantTemp2 & "')"

' Modificar la existencia en la tabla TelasColAlm, para cada Almacen
     Dbs.Execute "UPDATE TelasColAlm SET ExTela1 = ExTela1 - " & !CantTemp1 & ", ExTela2 = Extela2 - " & !CantTemp2 & " WHERE IdTelasColAlm = " & !IdTelasColAlmSalT
     Dbs.Execute "UPDATE TelasColAlm SET ExTela1 = ExTela1 + " & !CantTemp1 & ", ExTela2 = Extela2 + " & !CantTemp2 & " WHERE IdTelasColAlm = " & !IdTelasColAlmEntT

noagregar:
            .MoveNext
            
            
        Loop
        .Close
rstUltS.Close
rstUltE.Close

MsgBox "La transferencia de telas se realizo con exito.", vbInformation

Forms![ITelas_TransferAlm].ActualizarSalidas
Forms![ITelas_TransferAlm].ActualizarEntradas

Dbs.Execute ("DELETE * FROM ITelas_Temp")
Forms![ITelas_TransferAlm]![QueTela] = ""
Forms![ITelas_TransferAlm]![QueTela].SetFocus


End If
End With


Dbs.Close
Me.Requery


End If
