Private Sub UltimoNumero()
On Error GoTo errores
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Max(IdNotas) AS UltNum FROM Notas")
QueNota = Rst!ultnum
Rst.Close
Dbs.Close
Exit Sub
errores:
MsgBox Err.Description
End Sub




Private Sub Form_Open(Cancel As Integer)
If NivelAct > nivAdminDir Then
Modificar.Visible = False
End If
Call UltimoNumero
Me.Requery
End Sub
