Private Sub AltaMod_Click()
If IsNull(QueFecha) = True Then
MsgBox "Para poder continuar es necesario que metas una fecha.", vbInformation, "FALTA FECHA"
Hoya.SetFocus
Else

If QueModelo = "" Or IsNull(QueModelo) = True Or QueCant = "" Or IsNull(QueCant) = True Then
MsgBox "Para poder agregar un modelo es necesario que metas el modelo y la cantidad.", vbInformation, "FALTA INFORMACION"
Else
     If (QueExis - QueCant) < 0 Then
     MsgBox "OPERACION NO PERMITIDA." & vbCr & "No puedes dejar el modelo con existencia negativa!. Revisa bien la cantidad.", vbInformation, "OPERACION NO PERMITIDA!"
     QueCant.SetFocus
     Exit Sub
     End If

 
Dim Dbs As Database
Dim PrimerReg, TipoM, IdModAlmS, IdModAlmE, Rst As Recordset
Dim RefeSal, RefeEnt As String
Dim TipMov, IdMAS, IdMAE As Integer

Set Dbs = CurrentDb
Set TipoM = Dbs.OpenRecordset("SELECT IdIPT_TiposMov FROM IPT_TiposMov WHERE (TipoEnSa)=3")
TipMov = TipoM!IdIPT_TiposMov
If IsNull(ID_Salida) = True Then
  Dbs.Execute "INSERT INTO IPT_Movs (Fecha,IdIPT_TipoMov,EnSa,IdIPT_Almacenes,IdUsuarios,TieneModelos) VALUES ('" & QueFecha & "','" & TipMov & "','" & 2 & "','" & QueAlmS & "','" & IdUsuarioACT & "',1)"
  Dbs.Execute "INSERT INTO IPT_Movs (Fecha,IdIPT_TipoMov,EnSa,IdIPT_Almacenes,IdUsuarios,TieneModelos) VALUES ('" & QueFecha & "','" & TipMov & "','" & 1 & "','" & QueAlmE & "','" & IdUsuarioACT & "',1)"

  Set PrimerReg = Dbs.OpenRecordset("SELECT Max(IdIPT_Movs) as ullt FROM IPT_Movs")
  Id_Entrada = PrimerReg!ullt
  ID_Salida = Id_Entrada - 1
  PrimerReg.Close
  
  ' Actualizar Referencia
  RefeSal = AlmacenEn & " (" & Id_Entrada & ")"
  RefeEnt = AlmacenSa & " (" & ID_Salida & ")"
  Dbs.Execute "UPDATE IPT_Movs SET Referencia ='" & RefeSal & "' WHERE IdIPT_Movs = " & ID_Salida
  Dbs.Execute "UPDATE IPT_Movs SET Referencia ='" & RefeEnt & "' WHERE IdIPT_Movs = " & Id_Entrada
End If

' Ya se metieron los movimientos en la tabla IPT_Movs, ahora se van a meter los modelos en cada registro y se van a actualizar las existencias:
'Primero el movimiento de salida:
   Set IdModAlmS = Dbs.OpenRecordset("SELECT IdIPT_Mod_Alm FROM IPT_Mod_Alm WHERE IdIPT_Modelos=" & QueModelo & " and IdIPT_Almacenes=" & QueAlmS)
   IdMAS = IdModAlmS!IdIPT_Mod_Alm
   Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & ID_Salida & "','" & IdMAS & "','" & QueCant & "')"
   Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia - " & QueCant & " WHERE IdIPT_Mod_Alm = " & IdMAS
 
' Luego el de entrada:
   Set IdModAlmE = Dbs.OpenRecordset("SELECT IdIPT_Mod_Alm FROM IPT_Mod_Alm WHERE IdIPT_Modelos=" & QueModelo & " and IdIPT_Almacenes=" & QueAlmE)
   IdMAE = IdModAlmE!IdIPT_Mod_Alm
   Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & Id_Entrada & "','" & IdMAE & "','" & QueCant & "')"
   Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia + " & QueCant & " WHERE IdIPT_Mod_Alm = " & IdMAE

Set Rst = Dbs.OpenRecordset("SELECT Sum(CantMov) AS SumaCan FROM IPT_MovsDet GROUP BY IPT_MovsDet.IdIPT_Movs HAVING (IPT_MovsDet.IdIPT_Movs)=" & ID_Salida)


TipoM.Close
IdModAlmS.Close
IdModAlmE.Close

ListaSal.Requery
ListaEnt.Requery
CantMod = ListaSal.ListCount - 1
CantPiezas = Rst!sumacan
Rst.Close
Dbs.Close
QueModelo = ""
QueCant = ""
QueExis = ""
QueEmpresa = ""
QueModelo.SetFocus
End If
End If

Private Sub QueModelo_AfterUpdate()
Dim Dbs As Database
Dim Rst, RstExis As Recordset
On Error GoTo erss
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Empresa FROM IPT_Modelos INNER JOIN Empresas ON IPT_Modelos.IdEmpresas = Empresas.IdEmpresas WHERE (IPT_Modelos.IdIPT_Modelos)=" & QueModelo)
Set RstExis = Dbs.OpenRecordset("SELECT Existencia FROM IPT_Modelos INNER JOIN IPT_Mod_Alm ON IPT_Modelos.IdIPT_Modelos = IPT_Mod_Alm.IdIPT_Modelos WHERE (IPT_Modelos.IdIPT_Modelos)=" & QueModelo & " AND (IPT_Mod_Alm.IdIPT_Almacenes)=" & QueAlmS)
QueEmpresa = Rst!Empresa
QueExis = RstExis!Existencia
Rst.Close
RstExis.Close
Dbs.Close
Exit Sub
erss:

MsgBox "Se esta generando un error al dar de alta el modelo, no trates de dar de alta el modelo, y habla con Daniel Masri para ver que pasa.", vbInformation
 QueModelo = ""

End Sub
