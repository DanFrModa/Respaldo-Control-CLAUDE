Private Function QueFechaSeHizo(QueProc As Integer)
Dim Dbs As Database
Dim Prog As Recordset

Set Dbs = CurrentDb
Set Prog = Dbs.OpenRecordset("SELECT FechaReal FROM RC WHERE (IdOrdenes=" & IdOrdenes & ") AND NumProcesoRC=" & QueProc)
 If Prog.EOF = True Then
 QueFechaSeHizo = " NO LLEVA APLICACION"
 
Else
QueFechaSeHizo = Prog!FechaReal
       
End If

       
Prog.Close
Dbs.Close

End Function

Set Prog = Dbs.OpenRecordset("SELECT FechaEst FROM RC WHERE (IdOrdenes=" & IdOrdenes & ") AND NumProcesoRC=" & QueProc)

 QueFechaEst = " NO LLEVA APLICACION"

Prog.Close
Dbs.Close

End Function
Private Function QueResponsableEs(QueProc As Integer)
Dim Dbs As Database
Dim Prog As Recordset

Set Dbs = CurrentDb
Set Prog = Dbs.OpenRecordset("SELECT Nombre + ' ' + Apellidos AS QueNom FROM (Usuarios INNER JOIN RC_ProcUsua ON Usuarios.IdRC_TipoUsuarios = RC_ProcUsua.IdRC_TipoUsuarios) INNER JOIN CP_Procesos ON RC_ProcUsua.IdCP_Procesos = CP_Procesos.IdCP_Procesos WHERE (CP_Procesos.NumProceso=" & QueProc & ") ORDER BY RC_ProcUsua.IdRC_TipoUsuarios DESC")

If Prog.EOF = True Then
 QueResponsableEs = " NO LLEVA APLICACION"
 
Else
QueResponsableEs = Prog!QueNom
       
End If

Prog.Close
Dbs.Close

End Function
Private Function QueProcesoEs(QueProc As Integer)
Dim Dbs As Database
Dim Prog As Recordset

Set Dbs = CurrentDb
Set Prog = Dbs.OpenRecordset("SELECT Proceso FROM CP_Procesos WHERE NumProceso=" & QueProc)

If Prog.EOF = True Then
 QueProcesoEs = " NO LLEVA APLICACION"
 
Else
QueProcesoEs = Prog!Proceso
       
End If

Prog.Close
Dbs.Close

End Function
