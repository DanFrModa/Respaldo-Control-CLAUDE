Private Sub MeterInfo_Click()
On Error GoTo Err_MeterInfo_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "RC_MeterInfo"
    
    stLinkCriteria = "[IdOrdenes]=" & Me![IdOrdenes]
    DoCmd.OpenForm stDocName, , , stLinkCriteria
    
    Forms![RC_MeterInfo]![QueForm] = 2

Exit_MeterInfo_Click:
    Exit Sub

Err_MeterInfo_Click:
    MsgBox Err.Description
    Resume Exit_MeterInfo_Click
    
End Sub
