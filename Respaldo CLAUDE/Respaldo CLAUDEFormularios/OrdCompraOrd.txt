Private Sub Agregar_Click()
Dim Dbs As Database
Dim QueOrd As Recordset
Dim strmensaje As String

Set Dbs = CurrentDb
Set QueOrd = Dbs.OpenRecordset("SELECT IdOrdenes, Numero FROM Ordenes INNER JOIN PedidosDet ON Ordenes.IdPedidosDet = PedidosDet.IdPedidosDet WHERE (PedidosDet.IdPedidos)=" & QuePed & " ORDER BY Ordenes.Numero")
With QueOrd
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            strmensaje = "Orden a Agregar:" & !Numero & vbCr & "(Orden " & (.AbsolutePosition + 1) & " de " & .RecordCount & ")"
            If MsgBox(strmensaje, vbOKCancel, "AGREGAR ORDEN") = vbCancel _
                Then GoTo OtroMas
        
            Me![IdOrdenes] = !IdOrdenes
           
OtroMas:
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
End Sub
