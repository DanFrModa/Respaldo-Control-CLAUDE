Private Sub QueMaqVer_AfterUpdate()
If QueMaqVer = 4 Then
  CostSI = True
  ProcSi = True
ElseIf QueMaqVer = 2 Then
  CostSI = True
  ProcSi = False
ElseIf QueMaqVer = 3 Then
  CostSI = False
  ProcSi = True
End If
QueMaq.Requery
End Sub
