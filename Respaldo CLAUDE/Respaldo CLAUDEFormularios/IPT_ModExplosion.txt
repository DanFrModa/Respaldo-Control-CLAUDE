Option Compare Database

Private Sub AltaMod_Click()

Dim Dbs As Database
Dim Alm, Mo As Recordset

Set Dbs = CurrentDb
On Error GoTo errores

Set Mo = Dbs.OpenRecordset("Select IdIPT_Modelos FROM IPT_ModelosNoAlta")
Mo.MoveLast
Mo.MoveFirst


Set Alm = Dbs.OpenRecordset("Select IdIPT_Almacenes FROM IPT_Almacenes")
Alm.MoveLast
Alm.MoveFirst
   
   Do While Not Mo.EOF

        Do While Not Alm.EOF
             DoCmd.GoToRecord , , acNewRec
            IdIPT_Modelos = Mo!IdIPT_Modelos
            IdIPT_Almacenes = Alm!IdIPT_Almacenes
            Alm.MoveNext
        Loop
    Alm.MoveFirst
    Mo.MoveNext
   Loop

MsgBox "Los modelos se dieron de alta en los almacenes con exito.", vbInformation, "LISTO"
Mo.Close
Alm.Close

errores:
If Err.Number = 3021 Then
  MsgBox "No hay nuevos Modelos para dar de alta", vbInformation, "NO HAY NUEVOS MODELOS"
End If

Dbs.Close
End Sub
