Private Sub Alta_Telas_Click()
Dim Mens As String
Mens = "Estas Seguro que quieres dar de Alta un nuevo almacen de telas llamado '" & QueAlmacen & "', con todas las telas y colores que ya tienes dadas de alta en los otros almacenes? (Una vez creado, es imposible borrarolo)"
If MsgBox(Mens, vbYesNo, "Alta de Almacen") = vbYes Then

Me.RecordsetClone.FindFirst "[Almacen] = '" & Me![QueAlmacen] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
If Me!QueAlmacen = Me!Almacen Then
MsgBox "Ese nombre de almacen ya esta dado de alta. No es posible tener dos almacenes con el mismo nombre", vbExclamation, "ALMACEN"
GoTo ByeBye
Else

Dim Dbs As Database
Dim Almac, UltAlm As Recordset

Set Dbs = CurrentDb
Dbs.Execute "INSERT INTO Almacenes (Almacen) VALUES ('" & QueAlmacen & "')"

Set Almac = Dbs.OpenRecordset("SELECT [idtelascolores] FROM [telascolores]")
Set UltAlm = Dbs.OpenRecordset("SELECT Last(Almacenes.IdAlmacenes) AS Ult FROM Almacenes")
        
With Almac
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            IdAlmacenes = UltAlm!ult
            IdTelasColores = ![IdTelasColores]
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
MsgBox "Se dio de alta el nuevo almacen con exito", vbInformation, "LISTO"
DoCmd.Close
End If
End If
ByeBye:

End Sub
