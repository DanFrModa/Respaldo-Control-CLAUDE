Private Sub CopiarAnt_Click()
Dim Dbs As Database
Dim Rst As Recordset
Dim Ma As Variant

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdOrdenes, MaquilaOrd, AplicacionOrd FROM Ordenes WHERE IdOrdenes=" & IdOrdenes - 1)

MaquilaOrd = Rst!MaquilaOrd
AplicacionOrd = Rst!AplicacionOrd


Rst.Close
Dbs.Close
End Sub

Private Sub Siguiente_Click()


If FilterOn = True Then
 QueOrdenEstaba = IdOrdenes
 FilterOn = False
 Me.RecordsetClone.FindFirst "[Idordenes] = " & QueOrdenEstaba
    Me.Bookmark = Me.RecordsetClone.Bookmark
 
End If


On Error GoTo Err_Siguiente_Click


    DoCmd.GoToRecord , , acNext
    Call ActualizarOrden
    MaquilaOrd.SetFocus
    

Exit_Siguiente_Click:
    Exit Sub

Err_Siguiente_Click:
    MsgBox Err.Description
    Resume Exit_Siguiente_Click
    
End Sub
