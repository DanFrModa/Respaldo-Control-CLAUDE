Private Sub AltaMod_Click()
If IsNull(Fecha) = True Then
MsgBox "Para poder continuar es necesario que metas una fecha.", vbInformation, "FALTA FECHA"
Hoya.SetFocus
Else
  
If IsNull(QueModelo) = True Or IsNull(QueCant) = True Then
MsgBox "Para poder agregar un modelo es necesario que metas el modelo y la cantidad.", vbInformation, "FALTA INFORMACION"
Else

Dim Dbs As Database
Set Dbs = CurrentDb
  If EnSa = 1 Then
  Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & IdIPT_Movs & "','" & QueModelo & "','" & QueCant & "')"
  Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia + " & QueCant & " WHERE IdIPT_Mod_Alm = " & QueModelo
  TieneModelos = 1
  
  ElseIf EnSa = 2 Then
    If (QueExis - QueCant) < 0 Then
        If PrP(30) = True Then
           If MsgBox("Estas seguro de que quieres dejar este modelo con existencia negativa!.", vbYesNo, "MODELO EN NEGATIVO!") = vbYes Then
              Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & IdIPT_Movs & "','" & QueModelo & "','" & QueCant & "')"
              Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia - " & QueCant & " WHERE IdIPT_Mod_Alm = " & QueModelo
              TieneModelos = 1
           Else
           Exit Sub
           End If
        Else
        MsgBox "OPERACION NO PERMITIDA." & vbCr & "No puedes dejar el modelo con existencia negativa!. Revisa bien la cantidad.", vbInformation, "OPERACION NO PERMITIDA!"
        QueCant.SetFocus
        Exit Sub
        End If

    Else
        Dbs.Execute "INSERT INTO IPT_MovsDet (IdIPT_Movs,IdIPT_Mod_Alm,CantMov) VALUES ('" & IdIPT_Movs & "','" & QueModelo & "','" & QueCant & "')"
        Dbs.Execute "UPDATE IPT_Mod_Alm SET Existencia = Existencia - " & QueCant & " WHERE IdIPT_Mod_Alm = " & QueModelo
        TieneModelos = 1
    End If
  End If

Dim Rst As Recordset
Set Rst = Dbs.OpenRecordset("SELECT Sum(CantMov) AS SumaCan FROM IPT_MovsDet GROUP BY IPT_MovsDet.IdIPT_Movs HAVING (IPT_MovsDet.IdIPT_Movs)=" & IdIPT_Movs)


QueModLista.Requery
QueModelo = ""
QueCant = ""
QueExis = ""
QueEmpresa = ""
QueModelo.SetFocus
CantMod = QueModLista.ListCount - 1
CantPiezas = Rst!sumacan
Rst.Close
Dbs.Close
End If
End If

End Sub

Private Sub QueModelo_AfterUpdate()

Dim Rst, RstExis As Recordset

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Empresa FROM (IPT_Modelos INNER JOIN Empresas ON IPT_Modelos.IdEmpresas = Empresas.IdEmpresas) INNER JOIN IPT_Mod_Alm ON IPT_Modelos.IdIPT_Modelos = IPT_Mod_Alm.IdIPT_Modelos WHERE (IdIPT_Mod_Alm)=" & QueModelo)
Set RstExis = Dbs.OpenRecordset("SELECT Existencia FROM IPT_Mod_Alm WHERE (IdIPT_Mod_Alm)=" & QueModelo)
QueEmpresa = Rst!Empresa
QueExis = RstExis!Existencia

Rst.Close
RstExis.Close
Dbs.Close
Exit Sub
