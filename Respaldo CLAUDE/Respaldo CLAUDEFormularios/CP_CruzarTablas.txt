Private Sub Comando2_Click()


Dim Dbs As Database
Dim Art, Proc, NoProc As Recordset


Set Dbs = CurrentDb

Set NoProc = Dbs.OpenRecordset("Select CP_ProcNOAlta.IdCP_Procesos from CP_ProcNOAlta")
If NoProc.EOF = False Then
   MsgBox "Tienes que borrar los nuevos procesos antes de seguir con el alta de Nuevos Articulos.", vbInformation, "ERROR DE ALTA DE ARTICULOS"
 GoTo Adios
End If

On Error GoTo errores

Set Art = Dbs.OpenRecordset("Select CP_ArtNOAlta.IdCP_Articulos from CP_ArtNoAlta")
Art.MoveLast
Art.MoveFirst


Set Proc = Dbs.OpenRecordset("Select CP_Procesos.IdCP_Procesos from CP_Procesos")
Proc.MoveLast
Proc.MoveFirst
   
   Do While Not Art.EOF

        Do While Not Proc.EOF
             DoCmd.GoToRecord , , acNewRec
            IdCP_Articulos = Art!IdCP_Articulos
            IdCP_Procesos = Proc!IdCP_Procesos
            Proc.MoveNext
        Loop
    Proc.MoveFirst
    Art.MoveNext
   Loop

Art.Close
Proc.Close

errores:
If Err.Number = 3021 Then
  MsgBox "No hay nuevos articulos para dar de alta", vbExclamation, "NO HAY NUEVOS ARTICULOS"
End If
Adios:
Dbs.Close


End Sub
