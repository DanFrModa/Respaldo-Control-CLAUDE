Private Sub Form_Close()
If EstaCargado("EsMa_EdoCta") = True Then
 Forms![EsMa_EdoCta].Requery
End If
End Sub

Private Sub Form_Open(Cancel As Integer)
On Error GoTo erroress
If EstaCargado("EsMa_EdoCta") = True Then
    
    If Forms![EsMa_EdoCta]![QueTipoMaq] = 1 Then
     Me![EsMaRecibosSem].Form.RecordSource = "EsMaRecibosSemCon"
     EsEst = False
     
    ElseIf Forms![EsMa_EdoCta]![QueTipoMaq] = 2 Then
     Me![EsMaRecibosSem].Form.RecordSource = "EsMaRecibosSemEstCon"
     EsEst = True
     
    End If
    
  If IsNull(Forms![EsMa_EdoCta]![IdRecibos]) = False Then
    Me.RecordsetClone.FindFirst "[IdEsMa] = " & Forms![EsMa_EdoCta]![IdEsMa]
    Me.Bookmark = Me.RecordsetClone.Bookmark
  End If
End If
erroress:

End Sub

End Sub
