Private Sub Alta_Telas_Click()
Dim Mens As String
Mens = "Estas Seguro que quieres dar de Alta un nuevo Usuario llamado '" & QueUsuario & "', con todos los accesos? (Una vez creado, es imposible borrarolo)"
If MsgBox(Mens, vbYesNo, "Alta de Usuario") = vbYes Then

Me.RecordsetClone.FindFirst "[Usuario] = '" & Me![QueUsuario] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
If Me!QueUsuario = Me!Usuario Then
MsgBox "Ese nombre de usuario ya esta dado de alta. No es posible tener dos usuarios con el mismo nombre", vbExclamation, "USUARIO"
GoTo ByeBye
Else

Dim Dbs As Database
Dim Almac, UltAlm As Recordset

Set Dbs = CurrentDb
Dbs.Execute "INSERT INTO Usuarios (Usuario) VALUES ('" & QueUsuario & "')"

Set Almac = Dbs.OpenRecordset("SELECT [IdAccesos] FROM [Accesos]")
Set UltAlm = Dbs.OpenRecordset("SELECT Last(Usuarios.IdUsuarios) AS Ult FROM Usuarios")
        
With Almac
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            IdUsuarios = UltAlm!ult
            IdAccesos = ![IdAccesos]
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
MsgBox "Se dio de alta el nuevo USUARIO con exito. No olvides dar de alta toda la informacion", vbInformation, "LISTO"
DoCmd.Close
End If
End If
ByeBye:

End Sub
