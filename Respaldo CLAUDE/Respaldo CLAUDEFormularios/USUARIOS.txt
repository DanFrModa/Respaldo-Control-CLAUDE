Private Sub Verif()
Me.RecordsetClone.FindFirst "[Usuario] = '" & Me![QueUsuario] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
If Me!QueUsuario = Me!Usuario Then
  If Me!CantBloq >= 5 Then
      MsgBox ("Estas Bloqueado!!!, habla con Daniel Masri para que te arregle el problema")
  Else
    If Me!QueClave = Me!Clave Then
    CantBloq = 0
    Call VerifLog
    Call RegUsuarios
    Call AltaPropiedades
    NivelAct = Me!Nivel
    IdRC_TipoUsuariosAct = IdRC_TipoUsuarios
    IdUsuarioACT = Me!IdUsuarios
    DesdeQueAnoAct = DesdeQueAno
    
      If NivelAct <> 1 Then
      CambiarPropiedad "StartupShowDBWindow", dbBoolean, False
      CambiarPropiedad "AllowBuiltinToolbars", dbBoolean, False
      CambiarPropiedad "AllowFullMenus", dbBoolean, False
      CambiarPropiedad "AllowSpecialKeys", dbBoolean, False
      CambiarPropiedad "AllowBypassKey", dbBoolean, False
      CambiarPropiedad "AllowBreakIntoCode", dbBoolean, False
      CambiarPropiedad "AllowToolbarChanges", dbBoolean, False
      CambiarPropiedad "AllowShortcutMenus", dbBoolean, False
      
      Else
      
      CambiarPropiedad "StartupShowDBWindow", dbBoolean, True
      CambiarPropiedad "AllowBuiltinToolbars", dbBoolean, True
      CambiarPropiedad "AllowFullMenus", dbBoolean, True
      CambiarPropiedad "AllowSpecialKeys", dbBoolean, True
      CambiarPropiedad "AllowBypassKey", dbBoolean, True
      CambiarPropiedad "AllowBreakIntoCode", dbBoolean, True
      CambiarPropiedad "AllowToolbarChanges", dbBoolean, True
      CambiarPropiedad "AllowShortcutMenus", dbBoolean, True

      End If
      
      DoCmd.Close
      DoCmd.OpenForm "panel de control"
    
    Else
    MsgBox "Clave Incorrecta, Cuidado por que te puedes bloquear!!!", , "CLAVE INCORRECTA"
    CantBloq = CantBloq + 1
    DoCmd.GoToRecord , , acNext
    QueClave.SetFocus
    End If
  End If
Else
    MsgBox "Usuario no registrado. Si te quieres registrar pidele a Daniel Masri", , "USUARIO NO REGISTRADO"
    QueUsuario.SetFocus
End If
End Sub

Private Sub RegUsuarios()
Dim Dbs As Database
Dim Rst, Rst1 As Recordset
Set Dbs = CurrentDb
Dbs.Execute " INSERT INTO UsuariosLog (IdUsuarios) VALUES (" & IdUsuarios & ")"
Set Rst = Dbs.OpenRecordset(" SELECT Last(UsuariosLog.IdUsuariosLog) AS Ulti FROM UsuariosLog")
QueIdUsuariosLog = Rst!ulti
Rst.Close

Set Rst1 = Dbs.OpenRecordset("SELECT IdEmpresas, Importancia FROM Empresas WHERE Importancia=1")
EmpresaFav = Rst1!IdEmpresas
Rst1.Close
Dbs.Close
End Sub

Private Sub Seguridad()
         Dim i As Long
         Dim db As Database, RS As Recordset
         Dim RecordCount As Long
         Dim QueVariable As String
        On Error GoTo errores
         Set db = CurrentDb()
         Set RS = db.OpenRecordset(" SELECT UsuAccesos.IdAccesos, UsuAccesos.[Activado], UsuAccesos.IdUsuarios FROM UsuAccesos WHERE (UsuAccesos.IdUsuarios)=" & IdUsuarioACT & " ORDER BY UsuAccesos.IdAccesos")
        
         RS.MoveLast
         RecordCount = RS.RecordCount
     
         RS.MoveFirst
         For i = 1 To RecordCount
         QueVariable = CStr(i + 1)
            PrP(i) = RS![Activado]
            RS.MoveNext
         Next i
     
         For i = 1 To RecordCount
            Debug.Print PrP(i)
         Next i
         RS.Close
         db.Close
         Exit Sub
errores:
 If Err.Number = 3021 Then
 
 Else
 MsgBox "Tu usuario tiene un problema de seguridad. Habla con Daniel Masri para que te lo resuelva. Es probable que no funcione bien el sistema", vbCritical, "ERROR EN SISTEMA DE SEGURIDAD"
 End If

End Sub
