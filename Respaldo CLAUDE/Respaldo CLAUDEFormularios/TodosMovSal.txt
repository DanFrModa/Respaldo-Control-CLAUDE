Private Sub CargarFoto()
On Error GoTo Errorprueba

Dim Nu As Integer
Me!Imagen1.Picture = ""
Me!Imagen1.HyperlinkAddress = ""

Nu = 1
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
Else
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
End Select
End If
Resume Next
End Sub

Private Sub QueTela_AfterUpdate()

Dim Rst, Rst1 As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Corte.IdOrdenes, Sum(Corte.Cantidad) AS SumCant FROM Corte GROUP BY Corte.IdOrdenes HAVING (Corte.IdOrdenes)=" & IdOrdenes)
If Rst.EOF = True Then
CantCortada = 0
Else
CantCortada = Rst!sumcant
End If
Rst.Close

Set Rst1 = Dbs.OpenRecordset("SELECT Salidas.IdOrdenes, Sum([telasal1]+[telasal2]) AS TotalTel FROM Salidas INNER JOIN SalidasDet ON Salidas.IdSalidas = SalidasDet.IdSalidas GROUP BY Salidas.IdOrdenes HAVING (Salidas.IdOrdenes)=" & IdOrdenes)

Dbs.Close
Call CargarFoto
