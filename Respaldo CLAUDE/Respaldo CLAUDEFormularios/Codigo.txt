Option Compare Database
Option Explicit

Private Sub HacerCodigo()
Dim Texto1, Codi As Variant

If Opcion = 1 Then
 If Empresa = 1 Then
 Codi = "7500021"
 ElseIf Empresa = 2 Then
 Codi = "7509564"
 ElseIf Empresa = 3 Then
 Codi = "7500092"
 ElseIf Empresa = 4 Then
 Codi = "7500119"
 
 End If
 
Texto1 = Codi + Modelo
Else
Texto1 = CodigoBruto

End If

If Len(Texto1) <> 12 Or IsNull(Texto1) = True Then
MsgBox "Debes e meter 12 digitos"
Exit Sub
Else

Dim Total, Digito, Resultado, Resto, RestoDun, ResultadoDun As Integer
Dim Contador As Single

For Contador = 1 To 12
  Digito = Mid(Texto1, Contador, 1)
    If Contador Mod 2 = 0 Then
       Total = Total + (Digito * 3)
       Else
       Total = Total + (Digito * 1)
    End If
    
Next

Resto = Total Mod 10
RestoDun = (Total + 3) Mod 10

If Resto <> 0 Then
Resultado = 10 - Resto
End If

If RestoDun <> 0 Then
ResultadoDun = 10 - RestoDun
End If

CodigoCompleto = Texto1 + "" + Str(Resultado)
CodigoDUN = "1 " + Texto1 + "" + Str(ResultadoDun)

End If
End Sub

Private Sub CrearCodigo_Click()
Call HacerCodigo
End Sub
