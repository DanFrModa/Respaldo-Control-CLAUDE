Private Sub Form_Current()
Call QueEstatus
Me![OrdenVerSub].Form![TallasLoc] = Tallas
If OrdCancelada = True Then
CancEt.Visible = True
Else
CancEt.Visible = False
End If

On Error GoTo Errorprueba
QueError.Visible = False
Bord.Requery
Coment.Requery

If NivelAct > nivDirectivo Then
Else
If (IsNull(Costo) = True) Or (Costo = 0) Then
Costo.Visible = False
Else
Costo.Visible = True
End If
End If

If VerFotos = True Then
Dim Nu As Integer
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen3.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Me!Imagen3.HyperlinkAddress = ""
Nu = 1
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Nu = 2
Me!Imagen2.Picture = DirFotos(Foto2)
Me!Imagen2.HyperlinkAddress = DirFotos(Foto2)
Nu = 3
Me!Imagen3.Picture = DirBordados(FotoBord)
Me!Imagen3.HyperlinkAddress = DirBordados(FotoBord)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
QueError.Visible = True
QueError = Err.Description & ",Numero " & Err.Number
Else
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
Case 2
Me!Imagen2.Picture = NoFoto
Case 3
Me!Imagen3.Picture = NoFoto
End Select
End If
Resume Next
End If
End Sub

Private Function NombreCortoMaq()
Dim Dbs As Database
Dim Rst As Recordset
Dim Ma As Variant

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT First(Maquileros.Corto) AS NombreCorto, First(Maquileros.IdMaquileros) AS maquil FROM Entregas INNER JOIN Maquileros ON Entregas.IdMaquileros = Maquileros.IdMaquileros WHERE IdOrdenes =" & IdOrdenes)

Ma = Rst!nombrecorto
QueIdMaq = Rst!maquil
If IsNull(Ma) = True Then
NombreCortoMaq = "nel"
Else
NombreCortoMaq = Rst!nombrecorto
End If

Rst.Close
Dbs.Close

End Function
