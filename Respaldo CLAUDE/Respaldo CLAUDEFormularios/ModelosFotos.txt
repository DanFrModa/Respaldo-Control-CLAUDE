Private Sub Form_Current()
If Activo = True Then
Me!Act = "ACTIVO"
Me!Act.BackColor = vbGreen

ElseIf Activo = False Then
Me!Act = "INACTIVO"
Me!Act.BackColor = vbRed
End If

On Error GoTo Errorprueba
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)
Me!Imagen2.Picture = DirFotos(Foto2)
Me!Imagen2.HyperlinkAddress = DirFotos(Foto2)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
MsgBox Err.Description & vbCr & Err.Number
Else
Me!Imagen2.Picture = NoFoto
End If
End Sub

Private Sub Foto1_AfterUpdate()
On Error GoTo Errorprueba
Me!Imagen1.Picture = DirFotos(Foto1)
Me!Imagen1.HyperlinkAddress = DirFotos(Foto1)

Errorprueba:
If Err.Number <> 0 Then
MsgBox Err.Description
End If
End Sub

Private Sub Foto2_AfterUpdate()
On Error GoTo Errorprueba
Me!Imagen2.Picture = DirFotos(Foto2)
Me!Imagen2.HyperlinkAddress = DirFotos(Foto2)

Errorprueba:
If Err.Number <> 0 Then
MsgBox Err.Description
Me!Imagen2.Picture = NoFoto
End If
End Sub

Private Sub QueModelo_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdModelos] = " & Me![QueModelo]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub
Private Sub QueGenero_AfterUpdate()
If IsNull(QueGenero) = True Then
QueGeneroOculto = "*"
Else
QueGeneroOculto = QueGenero
End If
QueModeloLista.Requery
End Sub
Private Sub QueModeloLista_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdModelos] = " & Me![QueModeloLista]
    Me.Bookmark = Me.RecordsetClone.Bookmark
    QueModelo = QueModeloLista
End Sub
