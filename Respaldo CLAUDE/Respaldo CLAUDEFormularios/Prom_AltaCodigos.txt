Private Sub Verificar_Click()
If IsNull(QueModelo) = True Then
MsgBox "Tienes que meter un numero de modelo en el campo", vbExclamation
Else
    Me.RecordsetClone.FindFirst "[Modelo] = '" & Me![QueModelo] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
    
    If Me!QueModelo = Me!Modelo Then
    Call CargarFotos
    MsgBox "El modelo " & QueModelo & " ya esta dado de alta.", vbInformation, "MODELO YA EXISTENTE"
    
    Else
    If MsgBox("El modelo " & QueModelo & " no esta dado de alta. Quieres darlo de alta automaticamente?", vbYesNo, "MODELO NO EXISTENTE") = vbYes Then
    Dim Dbs As Database
    Set Dbs = CurrentDb
    Dbs.Execute "INSERT INTO Modelos (Modelo) VALUES ('" & QueModelo & "')"
    Dbs.Close
    MsgBox "Se dio de alta el modelo " & QueModelo, vbInformation, "ALTA MODELO " & QueModelo
    QueModelo = ""
        Me!Imagen1.Picture = ""
        Me!Imagen2.Picture = ""
        Me!Imagen1.HyperlinkAddress = ""
        Me!Imagen2.HyperlinkAddress = ""
    End If
    End If
End If
QueModelo.SetFocus
End Sub

Public Sub CargarFotos()
Dim Nu As Integer
On Error GoTo Errorprueba
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Nu = 1
Me!Imagen1.Picture = DirPromoda(QFoto1)
Me!Imagen1.HyperlinkAddress = DirPromoda(QFoto1)
Nu = 2
Me!Imagen2.Picture = DirPromoda(QFoto2)
Me!Imagen2.HyperlinkAddress = DirPromoda(QFoto2)
Exit Sub

Errorprueba:
Select Case Nu
Case 1
Me!Imagen1.Picture = NoFoto
Case 2
Me!Imagen2.Picture = NoFoto
End Select

Resume Next
End Sub



Private Sub VerificarDatos() '' Verifica Datos y genera el codigo y las fotos
Dim MenErr, Codi As String
Dim Erro As Boolean
Erro = False
MenErr = "Para poder generar el codigo, tienes que corregir los siguientes campos:" & vbCr

If IsNull(QProveedor) = True Then
MenErr = MenErr & vbCr & "- Debes de seleccionar un numero de Proveedor."
Erro = True
End If

If IsNull(QIdPrenda) = True Then
MenErr = MenErr & vbCr & "- Debes de seleccionar un Tipo de Prenda."
Erro = True
End If

If IsNull(QIdGenero) = True Then
MenErr = MenErr & vbCr & "- Debes de seleccionar un Genero."
Erro = True
End If

If IsNull(QVariasM) = True Then
MenErr = MenErr & vbCr & "- Debes de seleccionar si el codigo corresponde a un solo modelo o a varios modelos."
Erro = True
End If

If IsNull(QIdPartida) = True Then
MenErr = MenErr & vbCr & "- Debes de seleccionar una Partida."
Erro = True
End If

If IsNull(QPrecio) = True Then
MenErr = MenErr & vbCr & "- Debes de meter un precio."
Erro = True
End If


If Erro = True Then
MsgBox MenErr, vbExclamation, "FALTA INFORMACION"
Else
  

    Codi = QProveedor & Format(QIdPartida, "00") & Format(QPrecio, "000") & QIdPrenda & QIdGenero & Format(QConsecutivo, "000")
    If Len(Codi) = 13 Then
    QCodigo = Codi
    QFoto1 = [QIdPrenda] & [QIdGenero] & Format([QConsecutivo], "000") & "a"
    QFoto2 = [QIdPrenda] & [QIdGenero] & Format([QConsecutivo], "000") & "b"
    QModelo = [QIdPrenda] & [QIdGenero] & Format([QConsecutivo], "000")
  
    Call AgregarMod
    Else
    MsgBox "Hay algun error con el codigo!, no se esta pudiendo generar por alguna razon, por favor verifica la informacion.", vbExclamation, "ERROR EN GENERACION DE CODIGO"
    End If

End If
End Sub


Private Sub AgregarMod()
  Me.RecordsetClone.FindFirst "[ModeloProm] = '" & Me![QModelo] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
    
    If Me!QModelo = Me!ModeloProm Then
   
    MsgBox "El modelo " & QModelo & " ya esta dado de alta.", vbExclamation, "MODELO YA EXISTENTE"
    
    Else

On Error GoTo erss
Dim Dbs As Database
Set Dbs = CurrentDb
    Dbs.Execute "INSERT INTO Prom_Codigos (Proveedor,IdPrenda,IdGenero,Consecutivo,IdPartida,Precio,Foto1,Foto2,CostoUS,CantidadRec,IdUsuarios,CodigoProm,ModeloProm) VALUES ('" & QProveedor & "','" & QIdPrenda & "'," & QIdGenero & "," & QConsecutivo & "," & QIdPartida & "," & QPrecio & ",'" & QFoto1 & "','" & QFoto2 & "'," & QCostoUS & "," & QCantidadRec & "," & IdUsuarioACT & ",'" & QCodigo & "','" & QModelo & "')"
     
    Dbs.Execute "INSERT INTO Prom_CodigosTemp (CodigoTemp,ModeloTemp) VALUES ('" & QCodigo & "','" & QModelo & "')"
     
     
     If QVariasM = 1 Then
     Dbs.Execute "UPDATE Prom_Prenda SET ConsecUNO=ConsecUNO + 1 WHERE IdPrenda='" & QIdPrenda & "'"
     ElseIf QVariasM = 2 Then
     Dbs.Execute "UPDATE Prom_Prenda SET ConsecVARIOS=ConsecVARIOS + 1 WHERE IdPrenda='" & QIdPrenda & "'"
     ElseIf QVariasM = 3 Then
     Dbs.Execute "UPDATE Prom_Prenda SET ConsecENSALADA=ConsecENSALADA + 1 WHERE IdPrenda='" & QIdPrenda & "'"
     End If
Agregados.Requery
Dbs.Close
MsgBox "Se ha dado de alta el codigo con exito.", vbInformation, "SE AGREGO CON EXITO"
 Me.Requery
 End If
 Exit Sub
 
erss:
MsgBox Err.Description

End Sub
