Private Sub FillOptions()
' Llenar las opciones para esta página del panel de control.

    ' Cantidad de botones que hay en el formulario.
    
    Const conNumButtons = 8
    
    Dim Dbs As Database
    Dim Rst As Recordset
    Dim strSQL As String
    Dim intOption As Integer
    
    ' Establecer el enfoque al primer botón del formulario
    ' y ocultar todos los botones del formulario menos
    ' el primero. No se puede ocultar el campo que tiene
    ' el enfoque.
    Me![Option1].SetFocus
    For intOption = 2 To conNumButtons
        Me("Option" & intOption).Visible = False
        Me("OptionLabel" & intOption).Visible = False
    Next intOption
    
    ' Abrir la tabla Elementos del Panel de control y buscar
    ' el primer elemento para esta página del Panel de control.
    Set Dbs = CurrentDb()
    strSQL = "SELECT * FROM [Elementos del Panel de control nivel]"
    strSQL = strSQL & " WHERE [ItemNumber] > 0 AND [SwitchboardID]=" & Me![SwitchboardID]
    strSQL = strSQL & " ORDER BY [ItemNumber];"
    Set Rst = Dbs.OpenRecordset(strSQL)
    
    ' Si no hay opciones para esta página del Panel de control,
    ' mostrar un mensaje. Si no, llenar la página con los elementos.
    If (Rst.EOF) Then
        Me![OptionLabel1].Caption = "No hay elementos para esta página del Panel de control"
    Else
        While (Not (Rst.EOF))
            Me("Option" & Rst![ItemNumber]).Visible = True
            Me("OptionLabel" & Rst![ItemNumber]).Visible = True
            Me("OptionLabel" & Rst![ItemNumber]).Caption = Rst![ItemText]
            Rst.MoveNext
        Wend
    End If

    ' Cerrar el conjunto de registros y la base de datos.
    Rst.Close
    Dbs.Close

End Sub

Private Function HandleButtonClick(intBtn As Integer)
' Al hacer clic en un botón se llama a esta función.
' intBtn indica qué botón ha sido presionado.
 ''Call HayMant
    ' Constantes para los comandos que se pueden ejecutar.
    Const conCmdGotoSwitchboard = 1
    Const conCmdOpenFormAdd = 2
    Const conCmdOpenFormBrowse = 3
    Const conCmdOpenReport = 4
    Const conCmdCustomizeSwitchboard = 5
    Const conCmdExitApplication = 6
    Const conCmdRunMacro = 7
    Const conCmdRunCode = 8
    

    ' Un caso especial de error.
    Const conErrDoCmdCancelled = 2501
    
    Dim Dbs As Database
    Dim Rst As Recordset

On Error GoTo HandleButtonClick_Err

    ' Buscar en la tabla Elementos del Panel de control, el elemento
    ' que corresponde al botón presionado.
    Set Dbs = CurrentDb()
    Set Rst = Dbs.OpenRecordset("Elementos del Panel de control nivel", dbOpenDynaset)
    Rst.FindFirst "[SwitchboardID]=" & Me![SwitchboardID] & " AND [ItemNumber]=" & intBtn
    
    ' Si no corresponde ningún elemento, mostrar el error y salir
    ' de la función.
    If (Rst.NoMatch) Then
        MsgBox "Error al leer la tabla Elementos del Panel de control."
        Rst.Close
        Dbs.Close
        Exit Function
    End If
    
    Select Case Rst![Command]
        
        ' Ir a otro Panel de control.
        Case conCmdGotoSwitchboard
            UltMenu = Me!SwitchboardID
            Me.Filter = "[ItemNumber] = 0 AND [SwitchboardID]=" & Rst![Argument]
            
        ' Abrir un formulario en modo Agregar.
        Case conCmdOpenFormAdd
            DoCmd.OpenForm Rst![Argument], , , , acAdd

        ' Abrir un formulario.
        Case conCmdOpenFormBrowse
            DoCmd.OpenForm Rst![Argument]

        ' Abrir un informe.
        Case conCmdOpenReport
            DoCmd.OpenReport Rst![Argument], acPreview

        ' Personalizar el Panel de control.
        Case conCmdCustomizeSwitchboard
            ' Tratar el caso en que no está instalado el Administrador
            ' del Panel de control (por ejemplo, con la instalación
            ' mínima).
            On Error Resume Next
            Application.Run "WZMAIN80.sbm_Entry"
            If (Err <> 0) Then MsgBox "Este comando no está disponible."
            On Error GoTo 0
            ' Actualizar el formulario.
            Me.Filter = "[ItemNumber] = 0 AND [Argument] = 'Predeterminado' "
            Me.Caption = Nz(Me![ItemText], "")
            FillOptions

        ' Salir de la aplicación.
        Case conCmdExitApplication
            CloseCurrentDatabase

        ' Ejecutar una macro.
        Case conCmdRunMacro
            DoCmd.RunMacro Rst![Argument]

        ' Ejecutar código.
        Case conCmdRunCode
            Application.Run Rst![Argument]
            
        ' Yo lo hice... para abrir un formulario, y despues ir a un nuevo reg.
        Case 9
            DoCmd.OpenForm Rst![Argument]
            DoCmd.GoToRecord , , acNewRec
            
        ' Cualquier otro comando no se reconoce.
        Case Else
            MsgBox "Opción desconocida."
    
    End Select

    ' Cerrar el conjunto de registros y la base de datos.
    Rst.Close
    Dbs.Close
   
HandleButtonClick_Exit:
    Exit Function

HandleButtonClick_Err:
    ' Si el usuario canceló la acción por algún motivo,
    ' no mostrar ningún mensaje de error.
    ' Por el contrario, continuar en la línea siguiente.
    If (Err = conErrDoCmdCancelled) Then
        Resume Next
    Else
        MsgBox "Error al ejecutar el comando.", vbCritical
        Resume HandleButtonClick_Exit
    End If
    
End Function
