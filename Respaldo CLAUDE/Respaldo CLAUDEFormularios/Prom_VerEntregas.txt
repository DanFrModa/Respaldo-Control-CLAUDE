Private Sub DevCaja_Click()
If MsgBox("Estas seguro de querer marcar como devuelta la Caja Numero:" & Me![Prom_VerEntregasDet].Form![Caja] & " de la tienda No. " & Me![Prom_VerEntregasDet].Form![Tienda] & " ?", vbYesNo) = vbYes Then

Dim Dbs As Database
Dim Rst As Recordset

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdProm_EntregasDet FROM Prom_EntregasDet WHERE Caja=" & Me![Prom_VerEntregasDet].Form![Caja] & " And Tienda=" & Me![Prom_VerEntregasDet].Form![Tienda] & " And IdProm_Entregas=" & Me![Prom_VerEntregasDet].Form![IdProm_Entregas])
Rst.MoveLast
Rst.MoveFirst

Do While Not Rst.EOF
 Dbs.Execute "UPDATE Prom_EntregasDet SET  Devuelto = True WHERE IdProm_EntregasDet = " & Rst!IdProm_EntregasDet
 Rst.MoveNext
 Loop

Rst.Close
Dbs.Close
MsgBox ("Ya quedo marcada como devuelta la caja No." & Me![Prom_VerEntregasDet].Form![Caja] & " de la tienda No. " & Me![Prom_VerEntregasDet].Form![Tienda]), vbInformation
Me![Prom_VerEntregasDet].Requery
End If
End Sub

Private Sub DevTda_Click()
If MsgBox("Estas seguro de querer marcar como devueltas todas las cajas de la tienda No." & Me![Prom_VerEntregasDet].Form![Tienda] & " ?", vbYesNo) = vbYes Then

Dim Dbs As Database
Dim Rst As Recordset

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdProm_EntregasDet FROM Prom_EntregasDet WHERE Tienda=" & Me![Prom_VerEntregasDet].Form![Tienda] & " And IdProm_Entregas=" & Me![Prom_VerEntregasDet].Form![IdProm_Entregas])
Rst.MoveLast
Rst.MoveFirst

Do While Not Rst.EOF
 Dbs.Execute "UPDATE Prom_EntregasDet SET  Devuelto = True WHERE IdProm_EntregasDet = " & Rst!IdProm_EntregasDet
 Rst.MoveNext
 Loop

Rst.Close
Dbs.Close
MsgBox ("Ya quedo marcada como devueltas todas las cajas de la tienda No. " & Me![Prom_VerEntregasDet].Form![Tienda]), vbInformation
Me![Prom_VerEntregasDet].Requery

End If
End Sub

Public Sub CargarFotos()
On Error GoTo Errorprueba
Me!Imagen1.Picture = ""
Me!Imagen2.Picture = ""
Me!Imagen1.HyperlinkAddress = ""
Me!Imagen2.HyperlinkAddress = ""
Me!Imagen1.Picture = DirPromoda(QFoto1)
Me!Imagen1.HyperlinkAddress = DirPromoda(QFoto1)
Me!Imagen2.Picture = DirPromoda(QFoto2)
Me!Imagen2.HyperlinkAddress = DirPromoda(QFoto2)
Exit Sub

Errorprueba:
If Err.Number <> 94 Then
MsgBox Err.Description & vbCr & Err.Number
Else
Me!Imagen2.Picture = NoFoto
End If
End Sub
