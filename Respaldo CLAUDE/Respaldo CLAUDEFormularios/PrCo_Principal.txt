Private Sub SacarTela()
Dim Dbs As Database
  Dim RstTelas, RstModelos
  Set Dbs = CurrentDb
Set RstModelos = Dbs.OpenRecordset("SELECT Consumo FROM PrCo_Modelos WHERE (IdPrCo_Modelos)=" & Me!QueModelo)
Set RstTelas = Dbs.OpenRecordset("SELECT Precio, FactorPeso FROM PrCo_Telas WHERE (IdPrCo_Telas)=" & Me!QueTela)

CConsumo = RstModelos!Consumo * RstTelas!FactorPeso
CPrecioTela = RstTelas!Precio
CTela = (RstModelos!Consumo * RstTelas!FactorPeso) * (RstTelas!Precio)

RstModelos.Close
RstTelas.Close
Dbs.Close
End Sub
Private Sub SacarMaquila()

CMaquila = RstMaq!PrecioMaq

RstMaq.Close
Dbs.Close
End Sub
Private Function AumentarMargen(Precio As Single, queClie As Integer) As Single
Dim Tot As Single
Dim Dbs As Database
Dim Rst As Recordset
    Set Dbs = CurrentDb()
    Set Rst = Dbs.OpenRecordset("SELECT Descuento, MargenBruto From PrCo_Clientes WHERE IdPrCo_Clientes=" & queClie)
 
 Tot = ((Rst!Descuento + Rst!MargenBruto) - 1) * -1
    AumentarMargen = (Precio / Tot)
    
    Rst.Close
    Dbs.Close
End Function
Private Function PublicoSugerido(Venta As Single, queClie As Integer) As Single
Dim Tot, Tot1 As Single
Dim Dbs As Database
Dim Rst As Recordset
    Set Dbs = CurrentDb()
    Set Rst = Dbs.OpenRecordset("SELECT MargenEllos From PrCo_Clientes WHERE IdPrCo_Clientes=" & queClie)

    Tot = (Rst!MargenEllos - 1) * -1
    PublicoSugerido = Venta / Tot * 1.15
    
    Rst.Close
    Dbs.Close
End Function
