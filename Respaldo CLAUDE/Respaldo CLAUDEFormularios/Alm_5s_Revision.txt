Option Compare Database
Option Explicit

Private Sub Ayer_Click()


If IsNull(Hoy) = True Then
    If WeekDay(Date) = 2 Then
     FechaRevision_5s = Date - 3
    Else
       FechaRevision_5s = Date - 1
     End If
Else
  If MsgBox("Ya hay una fecha ingresada. Estas seguro de querer cambiarla?", vbYesNo, "YA HAY UNA FECHA INGRESADA") = vbYes Then
        If WeekDay(Date) = 2 Then
     FechaRevision_5s = Date - 3
    Else
       FechaRevision_5s = Date - 1
     End If
  End If
End If

End Sub

Private Sub Form_BeforeInsert(Cancel As Integer)
IdUsuarios = IdUsuarioACT
End Sub

Private Sub Form_Open(Cancel As Integer)
If PrP(16) = True Then
  FechaRevision_5s.Enabled = True
  FechaRevision_5s.Locked = False
  
  Me.DataEntry = False
  Me.QueFecha.Visible = True
End If
End Sub

Private Sub Hoy_Click()
If IsNull(Hoy) = True Then
  FechaRevision_5s = Date
Else
  If MsgBox("Ya hay una fecha ingresada. Estas seguro de querer cambiarla?", vbYesNo, "YA HAY UNA FECHA INGRESADA") = vbYes Then
    FechaRevision_5s = Date
  End If
End If
End Sub

Private Sub QueFecha_AfterUpdate()
 ' Buscar el registro que coincida con el control.
    Me.RecordsetClone.FindFirst "[IdAlm_5s] = " & Me![QueFecha]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub
