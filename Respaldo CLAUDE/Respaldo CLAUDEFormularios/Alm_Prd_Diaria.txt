CodeBehindForm
Option Compare Database
Option Explicit

Private Sub Ayer_Click()
If WeekDay(Date) = 2 Then
 FechaAlm = Date - 3
Else
 FechaAlm = Date - 1
End If
Call ChecarPersonal
End Sub





Private Sub Form_Current()
Call PermitirEntradas
End Sub

Private Sub Form_Open(Cancel As Integer)
If PrP(23) = False Then
 MsgBox "No estas autorizado para ingresar en esta pantalla. Si necesitas acceso, habla con Daniel Masri."
 DoCmd.Close
 Exit Sub
End If

If WeekDay(Date) = 2 Then
 Sabado.Visible = True
End If

If PrP(16) = True Then
QueFecha.Visible = True
Me.DataEntry = False
FechaAlm.Enabled = True
FechaAlm.Locked = False
End If

End Sub


Private Sub HorasTrabajadas_AfterUpdate()
Call PermitirEntradas
End Sub

Private Sub Hoya_Click()
FechaAlm = Date
Call ChecarPersonal
End Sub

Private Sub FechaAlm_AfterUpdate()
Call ChecarPersonal
End Sub

Private Sub QuePersonal_AfterUpdate()
Call PermitirEntradas
End Sub

Private Sub Personas_AfterUpdate()
Call PermitirEntradas
End Sub

Private Sub QueFecha_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdAlm_Prd] = " & Me![QueFecha]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub Sabado_Click()
 QueFecha = Date - 2
 Call ChecarPersonal
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub

Private Sub PermitirEntradas()
If (IsNull(FechaAlm) = False) And (Personas <> 0) And (HorasTrabajadas <> 0) Then
  Me![Alm_Prd_Detalle].Form.AllowAdditions = True
Else
 Me![Alm_Prd_Detalle].Form.AllowAdditions = False
End If
End Sub

Private Sub ChecarPersonal()
If IsNull(Personas) = True Or Personas = 0 Then
 Personas.SetFocus
Else
 Call PermitirEntradas
End If
End Sub
Private Sub NUevoDia_Click()
On Error GoTo Err_NUevoDia_Click


    DoCmd.GoToRecord , , acNewRec

Exit_NUevoDia_Click:
    Exit Sub

Err_NUevoDia_Click:
    MsgBox Err.Description
    Resume Exit_NUevoDia_Click
    
End Sub
