CodeBehindForm
Option Compare Database
Option Explicit
Dim Ca As Integer

Private Sub AbrirRecibo_Click()

If IsNull(IdMaquileros) = True Then
 MsgBox "Debes de registrar el maquilero al que se le va a entregar para poder continuar.", vbInformation
 
Else
Dim QueIdEnt As Integer
QueIdEnt = IdEntregas

DoCmd.GoToRecord , , acNext
DoCmd.GoToRecord , , acPrevious

DoCmd.OpenForm "OrdDetEntregas", , , "[IdEntregas]=" & QueIdEnt
Forms![OrdDetEntregas]![OrdDetEntM].Form![TallasLoc] = Forms![Proceso]![Tallas]
Forms![OrdDetEntregas]![OrdDetEntM].Form![QueIdOrdenes] = IdOrdenes
Forms![OrdDetEntregas]![OrdDetEntM].Form![IdOrdenesDet].Requery
  
  If Forms![OrdDetEntregas]![PrecioPactado] = 0 Or IsNull(Forms![OrdDetEntregas]![PrecioPactado]) = True Then
   Forms![OrdDetEntregas]![PrecioPactado] = Forms![Proceso]![MaquilaOrd]
  End If

End If

End Sub

Private Sub Cant_Click()
DoCmd.OpenForm ("OrdDetEntM")
Forms![OrdDetEntM]![QueIdOrdenes] = IdOrdenes
Forms![OrdDetEntM]![QueIdEntregas] = IdEntregas
Forms![OrdDetEntM]![TallasLoc] = Forms![Proceso]![Tallas]
Forms![OrdDetEntM].BorrarAa
Forms![OrdDetEntM]![IdOrdenesDet].Requery
End Sub

Private Sub Cantidad_AfterUpdate()
If Forms![Proceso]![TotEntrega] + Me!Cantidad - Ca > Forms![Proceso]![TotCortada] Then
MsgBox "No puede ser mayor la cantidad total entregada al maquilero que la cantidad total cortada!!!. Checa bien el dato."
End If
End Sub

Private Sub Cantidad_GotFocus()
Ca = Me!Cantidad
End Sub

Private Sub Cantidad_LostFocus()
If Me!Cantidad = 0 Then
Me!Cantidad = Forms![Proceso]![TotCortada]
End If
End Sub

Private Sub Form_AfterInsert()
Forms![Proceso]![ProcesoRecibo].Form![Maquilero].Requery
End Sub

Private Sub Form_BeforeInsert(Cancel As Integer)

Dim Rst As Recordset
Set Rst = CurrentDb.OpenRecordset("SELECT Max(Consecutivo) as MaxCons FROM Entregas WHERE IdOrdenes=" & Forms![Proceso]![IdOrdenes])
If IsNull(Rst!maxcons) = True Or Rst!maxcons = 0 Then
 Consecutivo = 1
Else
 Consecutivo = Rst!maxcons + 1
End If
End Sub


Private Sub Imp_Click()
On Error GoTo errores
DoCmd.OpenForm "ReciboMaquilaImp", acPreview, , "[IdEntregas]=" & Me![IdEntregas]

Exit Sub
errores:
MsgBox Err.Description

End Sub

Private Sub Maquilero_LostFocus()
If IsNull(Maquilero) = True Then
Maquilero = Forms![Proceso]![IdMaquileros]
End If
End Sub
