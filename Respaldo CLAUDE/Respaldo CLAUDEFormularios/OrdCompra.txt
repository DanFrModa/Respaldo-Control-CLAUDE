Option Compare Database
Option Explicit

Private Sub AumentarNumNota()
On Error GoTo errores
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Max(OrdCompra.NumCompra) AS UltNum FROM OrdCompra")
Me!NumCompra = Rst!ultnum + 1
Rst.Close
Dbs.Close
Exit Sub
errores:
MsgBox Err.Description
End Sub

Private Sub Catalogo_Click()
DoCmd.OpenForm "Proveedores"
End Sub

Private Sub Comando43_Click()
If IsNull(IdOrdCompraExcel) = True Then
 If MsgBox("Estas seguro de que quieres crear una hoja de Excel, para meter el desglose de la cantidades de la orden de compra?", vbYesNo, "DESGLOSE") = vbYes Then
 DoCmd.OpenForm "OrdCompraExcel", , , , acFormAdd
 Forms![OrdCompraExcel]![IdOrdCompra] = Me!IdOrdCompra
 Forms![OrdCompraExcel]![NumeroCom] = NumCompra
 End If
Else
MsgBox "Esta orden de compra ya tiene una hoja de excel con el desglose. No es posible meter dos para la misma orden. Si quieres modificarla, vete a modificar orden desde la consulta de las ordenes", vbInformation

End If
End Sub

Private Sub Form_BeforeInsert(Cancel As Integer)
If MsgBox("Con esta accion, se va a generar una nueva orden de compra, y se le va a asignar un nuevo numero. Estas seguro de que quieres meter una nueva orden de compra?", vbYesNo, "Generar Orden de compra") = vbYes Then
Call AumentarNumNota
IdUsuarios = IdUsuarioACT
If NivelAct <= nivGerencial Then
Autorizado = True
IdUsuAutorizado = IdUsuarioACT
FechaAutorizado = Now()
End If
Else
DoCmd.Close
End If
End Sub

Private Sub Form_Close()
If EstaCargado("OrdCompraVer") = True Then
Forms![OrdCompraVer].Requery
End If
End Sub

Private Sub Form_Current()

Me![OrdCompraDet].Form![CantTotal].Visible = False
Else
Me![OrdCompraDet].Form![CantTotal].Visible = True
End If
End Sub
