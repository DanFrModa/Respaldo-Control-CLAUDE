Private Sub AltaMod_Click()
Dim Mens As String
Dim UltIdMod As Integer
Mens = "Estas Seguro que quieres dar de Alta un nuevo modelo en los almacenes de Prenda Terminada con el Numero de Modelo " & QueNumMod & ",? (Una vez creado, es imposible borrarlo)"
If MsgBox(Mens, vbYesNo, "Alta de Modelo") = vbYes Then

Me.RecordsetClone.FindFirst "[nummod] = '" & Me![QueNumMod] & "'"
    Me.Bookmark = Me.RecordsetClone.Bookmark
 If Me!QueNumMod = Me!NumMod Then
   If MsgBox("Ese numero de Modelo ya esta dado de alta. Estas Seguro de querer dar de alta otro numero de modelo identico?", vbYesNo, "MODELO EXISTENTE") = vbNo Then
     GoTo ByeBye
   End If
 End If

Dim Dbs As Database
Dim Almac, UltMod As Recordset

Set Dbs = CurrentDb
Dbs.Execute "INSERT INTO IPT_Modelos (NumMod) VALUES ('" & QueNumMod & "')"

Set Almac = Dbs.OpenRecordset("SELECT IdIPT_Almacenes FROM IPT_Almacenes")
Set UltMod = Dbs.OpenRecordset("SELECT Max(IdIPT_Modelos) AS Ult FROM IPT_Modelos")
 UltIdMod = UltMod!ult
With Almac
        .MoveLast
        .MoveFirst

        Do While Not .EOF
             DoCmd.GoToRecord , , acNewRec
            IdIPT_Modelos = UltIdMod
            IdIPT_Almacenes = ![IdIPT_Almacenes]
            .MoveNext

        Loop

        .Close
    End With
    
Dbs.Close
QueNumMod = ""
QueNumMod.SetFocus
MsgBox "Se dio de alta el nuevo Modelo con exito. Ahora solo debes de clasificarlo", vbInformation, "LISTO"
Call ActForms
DoCmd.OpenForm "IPT_Modelos", , , "[IdIPT_Modelos]=" & UltIdMod
 
Forms![IPT_Modelos]![QueModelo].Visible = False
Forms![IPT_Modelos]![Bloq].Visible = False
 End If

ByeBye:

End Sub

Call ActForms
