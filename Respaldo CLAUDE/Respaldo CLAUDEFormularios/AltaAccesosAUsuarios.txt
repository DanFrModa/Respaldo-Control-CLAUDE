Private Sub Comando2_Click()

Dim Dbs As Database
Dim Acc, Usu As Recordset


Set Dbs = CurrentDb


On Error GoTo errores

Set Acc = Dbs.OpenRecordset("Select IdAccesos from AccNOAlta")
Acc.MoveLast
Acc.MoveFirst


Set Usu = Dbs.OpenRecordset("Select IdUsuarios FROM Usuarios")
Usu.MoveLast
Usu.MoveFirst
   
   Do While Not Acc.EOF

        Do While Not Usu.EOF
             DoCmd.GoToRecord , , acNewRec
            IdAccesos = Acc!IdAccesos
            IdUsuarios = Usu!IdUsuarios
            Usu.MoveNext
        Loop
    Usu.MoveFirst
    Acc.MoveNext
   Loop

Acc.Close
Usu.Close

errores:
If Err.Number = 3021 Then
  MsgBox "No hay nuevos Accesos para dar de alta", vbExclamation, "NO HAY NUEVOS ACCESOS"
End If

Dbs.Close


End Sub
