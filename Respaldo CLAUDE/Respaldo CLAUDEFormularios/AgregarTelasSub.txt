Private Sub Form_AfterInsert()
Dim Dbs As Database
Set Dbs = CurrentDb
Dim Almac As Recordset
Dim Aa As String

Set Almac = Dbs.OpenRecordset("SELECT [idalmacenes] FROM [almacenes]")
Aa = Forms![AgregarTelas]![AgregarTelasSub].Form![IdTelasColores]

With Almac
        .MoveLast
        .MoveFirst

        Do While Not .EOF
    Dbs.Execute ("INSERT INTO [TelasColAlm] (IdAlmacenes,IdTelasColores) VALUES (" & !IdAlmacenes & "," & Aa & ")")
    
            .MoveNext

        Loop

        .Close
    End With



Dbs.Close
End Sub
