Private Sub CopiarAnt_Click()
Dim Dbs As Database
Dim Rst As Recordset
Dim Ma As Variant

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdOrdenes, IdCP_Articulos, IdRC_Aplicaciones, IdRC_TipoTelas, SI_RC, RC_Viva FROM Ordenes WHERE IdOrdenes=" & IdOrdenes - 1)

IdCP_Articulos = Rst!IdCP_Articulos
IdRC_Aplicaciones = Rst!IdRC_Aplicaciones
IdRC_TipoTelas = Rst!IdRC_TipoTelas
SI_RC = Rst!SI_RC
RC_Viva = Rst!RC_Viva


Rst.Close
Dbs.Close
End Sub

Private Sub Form_Close()
On Error GoTo Errs
If QueForm = 1 Then
 Forms![OrdenVer].Requery
 Forms![OrdenVer].RecordsetClone.FindFirst "[Idordenes] = " & Forms![OrdenVer]![QueOrden]
 Forms![OrdenVer].Bookmark = Forms![OrdenVer].RecordsetClone.Bookmark
 
ElseIf QueForm = 2 Then
 Forms![RC_Programacion].Requery
 Forms![RC_Programacion].RecordsetClone.FindFirst "[Idordenes] = " & Forms![RC_Programacion]![QueOrden]
 Forms![RC_Programacion].Bookmark = Forms![RC_Programacion].RecordsetClone.Bookmark
End If
Errs:

End Sub

Private Sub Form_Open(Cancel As Integer)
If EstaCargado("OrdenVer") = True Then
 Siguiente.Visible = True
 Anterior.Visible = True
 CopiarAnt.Visible = True
End If
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub
Private Sub Siguiente_Click()

If FilterOn = True Then
 QueOrdenEstaba = IdOrdenes
 FilterOn = False
 Me.RecordsetClone.FindFirst "[Idordenes] = " & QueOrdenEstaba
    Me.Bookmark = Me.RecordsetClone.Bookmark
 
End If



On Error GoTo Err_Siguiente_Click


    DoCmd.GoToRecord , , acNext
    
    If EstaCargado("ordenver") = True Then
    Call ActualizarOrden
    End If

Exit_Siguiente_Click:
    Exit Sub

Err_Siguiente_Click:
    MsgBox Err.Description
    Resume Exit_Siguiente_Click
    
End Sub
