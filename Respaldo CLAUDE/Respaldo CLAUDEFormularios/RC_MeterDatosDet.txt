Private Sub FechaReal_AfterUpdate()
If IsNull(IdUsuarioACT) = True Or IdUsuarioACT = 0 Then
   MsgBox "No tienes un usuario valido para meter la fecha. Salte de CONTROL y vuelvete a meter para continuar", vbExclamation, "USUARIO INVALIDO"
   FechaReal = ""
   Exit Sub
  End If

If PrP(10) = True Then
   Me.IdUsuario = IdUsuarioACT
   Me.FechaUsuarioRC = Now()
Else

  Dim QueFechaHoy As Date
  If WeekDay(Date) = vbMonday Then
    QueFechaHoy = Date - 2
  ElseIf WeekDay(Date) = vbTuesday Then
    QueFechaHoy = Date - 2
  Else
    QueFechaHoy = Date
  End If

  If FechaReal + 2 < QueFechaHoy Then
   FechaReal = ""
   MsgBox "No puedes meter una fecha con mas de dos dias de retrazo. Para poder meterla, pidele al Administrador.", vbInformation, "RETRAZO EN INGRESO DE INFORMACION"
   FechaReal.SetFocus
  Else
  Me.IdUsuario = IdUsuarioACT
  Me.FechaUsuarioRC = Now()

  Call MatarRC
  End If

End If
End Sub

End Sub

Private Sub Form_Current()
If IsNull(NumProcesoRC) = True Then
 FechaReal.Enabled = False
Else
 FechaReal.Enabled = True
End If
End Sub

Private Sub Hoy_Click()
If IsNull(IdUsuarioACT) = True Or IdUsuarioACT = 0 Then
   MsgBox "No tienes un usuario valido para meter la fecha. Salte de CONTROL y vuelvete a meter para continuar", vbExclamation, "USUARIO INVALIDO"
   FechaReal = ""
   Exit Sub
  End If

FechaReal = Date
Me.IdUsuario = IdUsuarioACT
Me.FechaUsuarioRC = Now()
Call MatarRC
End Sub

Private Sub MatarRC()
If Forms![RC_MeterFechas]![UltimoProceso] = True Then

Dim Dbs As Database
Set Dbs = CurrentDb

Dbs.Execute ("UPDATE Ordenes Set Ordenes.RC_Viva = False Where IdOrdenes=" & Forms![RC_MeterFechas]![RC_MeterFechasSub].Form![IdOrdenes])
Dbs.Close
End If
End Sub
