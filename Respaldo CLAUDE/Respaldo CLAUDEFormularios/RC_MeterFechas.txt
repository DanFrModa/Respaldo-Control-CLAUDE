Private Sub Form_Current()
Select Case TipoProceso
 Case "L"
  Me![RC_MeterDatosDet].Form.Visible = True
  Me![RC_IP3].Form.Visible = False
  Me![RC_IP4].Form.Visible = False
  Me![RC_IP5].Form.Visible = False
  
 Case "M"
  Me![RC_MeterDatosDet].Form.Visible = False
  Me![RC_IP3].Form.Visible = True
  Me![RC_IP4].Form.Visible = False
  Me![RC_IP5].Form.Visible = False
  
 Case "F"
  Me![RC_MeterDatosDet].Form.Visible = False
  Me![RC_IP3].Form.Visible = False
  Me![RC_IP4].Form.Visible = True
  Me![RC_IP5].Form.Visible = False
  
 Case "MP"
  Me![RC_MeterDatosDet].Form.Visible = False
  Me![RC_IP3].Form.Visible = False
  Me![RC_IP4].Form.Visible = False
  Me![RC_IP5].Form.Visible = True
 Case Else
  Me![RC_MeterDatosDet].Form.Visible = True
  Me![RC_IP3].Form.Visible = False
  Me![RC_IP4].Form.Visible = False
  Me![RC_IP5].Form.Visible = False
End Select
End Sub

Private Sub Form_Load()
On Error GoTo ErS
Me![RC_MeterDatosDet].Form.RecordSource = "SELECT RC.IdRC, RC.NumProcesoRC, RC.AntecesorRC, RC.IdUsuario,RC.FechaUsuarioRC,RC.FechaReal, Ordenes.Numero, Modelos.Modelo, CP_Procesos.NumProceso, CP_Procesos.Proceso, Clientes.Cliente, RC_Viva FROM (((RC LEFT JOIN Ordenes ON RC.IdOrdenes = Ordenes.IdOrdenes) LEFT JOIN Modelos ON Ordenes.IdModelos = Modelos.IdModelos) LEFT JOIN CP_Procesos ON RC.IdCP_Procesos = CP_Procesos.IdCP_Procesos) LEFT JOIN Clientes ON Ordenes.IdClientes = Clientes.IdClientes WHERE (((RC.IdRC)=[Formularios]![RC_MeterFechas]![RC_MeterFechasSub].[Formulario]![IdRC]) and RC_Viva=true)"
ErS:
End Sub
