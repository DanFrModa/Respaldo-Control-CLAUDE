Option Compare Database
Option Explicit


Private Sub Form_BeforeInsert(Cancel As Integer)
If MsgBox("Con esta accion, se va a generar una nueva Auditoria, y se le va a asignar un nuevo numero. Estas seguro de que quieres meter una nueva Auditoria?", vbYesNo, "Generar Orden de compra") = vbYes Then
Call AumentarNumAudit
Call InsertarFav
IdUsuariosElaboro = IdUsuarioACT
Else
DoCmd.Close
End If
End Sub

Private Sub AumentarNumAudit()
On Error GoTo errores
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Max(CC_Auditorias.NumAuditoria) AS UltNum FROM CC_Auditorias")
Me!NumAuditoria = Rst!ultnum + 1
Rst.Close
Dbs.Close
Exit Sub
errores:
MsgBox Err.Description
End Sub

Private Sub TraerCant()
On Error GoTo Era
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdOrdenes,CantOrd FROM CantidadOrd WHERE IdOrdenes=" & Me.IdOrdenes)
Me.Cant = Rst!CantOrd
Rst.Close
Dbs.Close
Era:
End Sub

Private Sub Form_Current()
If PrP(11) = True Then
IdMaquilero.Requery
End If
Call TraerCant
Call PrimerMaq
End Sub

Private Sub Form_Open(Cancel As Integer)
If PrP(11) = False Then
 MsgBox "No estas autorizado para generar Auditorias. Si necesitas hacer esto, habla con Daniel Masri.", vbInformation, "NO AUTORIZADO"
 DoCmd.Close
Else

  If EstaCargado("CC_ConsultaAuditorias") = True Then
    Me.RecordsetClone.FindFirst "[IdCC_Auditorias] = " & Forms![CC_ConsultaAuditorias]![IdCC_Auditorias]
    Me.Bookmark = Me.RecordsetClone.Bookmark
  ElseIf EstaCargado("CC_ConsulAuditMaq") = True Then
     Me.RecordsetClone.FindFirst "[IdCC_Auditorias] = " & Forms![CC_CC_ConsulAuditMaq]![IdCC_Auditorias]
    Me.Bookmark = Me.RecordsetClone.Bookmark
  Else
    DoCmd.GoToRecord , , acNewRec
  End If
End If

If PrP(12) = True Then
QueAuditoria.Visible = True
End If


End Sub

Private Sub IdOrdenes_AfterUpdate()
IdMaquilero.Requery
Call TraerCant
Call PrimerMaq
End Sub


Private Sub InsertarFav()
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdCC_Catalogo, Favorito FROM CC_Catalogo WHERE Favorito=True")

Rst.MoveLast
Rst.MoveFirst
  
 Do While Not Rst.EOF
       Dbs.Execute "INSERT INTO CC_AuditoriasDet (IdCC_Auditorias,IdCC_Catalogo) VALUES (" & Me.IdCC_Auditorias & "," & Rst!IdCC_Catalogo & ")"

            Rst.MoveNext
     Loop


Rst.Close
Dbs.Close
Me![CC_AltaAuditDet].Form.Requery
Me![CC_AltaAudCat].Form.Requery

End Sub

Private Sub QueAuditoria_AfterUpdate()
    Me.RecordsetClone.FindFirst "[IdCC_Auditorias] = " & Me![QueAuditoria]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub

Private Sub PrimerMaq()
On Error GoTo Era
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdOrdenes,IdMaquileros FROM Entregas WHERE IdOrdenes=" & Me.IdOrdenes)
Me.IdMaquilero = Rst!IdMaquileros
Rst.Close
Dbs.Close
Era:
End Sub
