Private Sub Form_Current()
Me![CostoOrdSub].Form![HabCalc] = 0
On Error GoTo errores
    Me![CostoOrdSub].Form![TelaLista].Requery
    Me![CostoOrdSub].Form![HabLista].Requery
    Me![CostoOrdSub].Form![BorLista].Requery
  Dim Dbs As Database
  Dim RstTelas, RstHab, RstReg As Recordset
  Set Dbs = CurrentDb
Set RstTelas = Dbs.OpenRecordset("SELECT Salidas.IdOrdenes, Sum(SalidasDet.TelaSal1) AS TotTela1, Sum(SalidasDet.TelaSal2) AS TotTela2, Sum(([telasal1]+[telasal2])*[precio]) AS Importe FROM TelasColores INNER JOIN ((SalidasDet INNER JOIN Salidas ON SalidasDet.IdSalidas = Salidas.IdSalidas) INNER JOIN TelasColAlm ON SalidasDet.IdTelasColAlm = TelasColAlm.IdTelasColAlm) ON TelasColores.IdTelasColores = TelasColAlm.IdTelasColores GROUP BY Salidas.IdOrdenes HAVING (Salidas.IdOrdenes)=" & Me!IdOrdenes)
 
  Me![CostoOrdSub].Form![TotTela] = RstTelas!TotTela1 + RstTelas!ToTTela2
  Me![CostoOrdSub].Form![ImporteTotal] = RstTelas!Importe
  Me![CostoOrdSub].Form![TelaCalc] = RstTelas!Importe / Me!CantCorte
RstTelas.Close
  
Set RstHab = Dbs.OpenRecordset("SELECT Sum([canthabord]*[preciohabord]) AS ImporteHab, OrdenesHab.IdOrdenes FROM Habilitacion INNER JOIN OrdenesHab ON Habilitacion.IdHabilitacion = OrdenesHab.IdHabilitacion GROUP BY OrdenesHab.IdOrdenes HAVING (OrdenesHab.IdOrdenes)=" & Me!IdOrdenes)

  Me![CostoOrdSub].Form![HabCalc] = RstHab!Importehab
RstHab.Close
  
  
Set RstReg = Dbs.OpenRecordset("SELECT Ordenes.IdOrdenes, [precio]*[regalias] AS CostoRegalias FROM (PedidosDet INNER JOIN Ordenes ON PedidosDet.IdPedidosDet = Ordenes.IdPedidosDet) INNER JOIN EtiquetasM ON Ordenes.IdEtiquetasM = EtiquetasM.IdEtiquetasM WHERE (Ordenes.IdOrdenes)=" & Me!IdOrdenes)
Me![CostoOrdSub].Form![RegaliasCalc] = RstReg!CostoRegalias
RstReg.Close
  
  Dbs.Close
  Exit Sub
errores:
 '' MsgBox Err.Description
  
End Sub
