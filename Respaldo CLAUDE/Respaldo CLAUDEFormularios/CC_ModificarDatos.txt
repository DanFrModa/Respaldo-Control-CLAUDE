CodeBehindForm
Option Compare Database
Option Explicit

Private Sub AgragarObs_Click()
If MsgBox("Con esta accion se va a agregar la informacion que esta en el cuadro de Agregar Observaciones a las observaciones de esta auditoria, y es imposible borrarlo. Estas Seguro de querer hacer esto?", vbYesNo, "Agregar mas Observaciones") = vbYes Then
  Me!Observaciones = Me!Observaciones & vbNewLine & QueObser
  QueObser = ""
  Me.Refresh
End If
End Sub

Private Sub CancelarAudit_Click()
If MsgBox("Estas seguro de querer cancelar la Auditoria " & NumAuditoria & "?", vbYesNo, "CANCELAR AUDITORIA") = vbYes Then
Cancelada = True
Salir.SetFocus
CancelarAudit.Visible = False
CancelEtiqueta.Visible = True
End If
End Sub

Private Sub Form_Current()
IdMaquilero.Requery
QueObser = ""

If Cancelada = True Then
CancelarAudit.Visible = False
CancelEtiqueta.Visible = True
Else
CancelarAudit.Visible = True
CancelEtiqueta.Visible = False
End If

End Sub


Private Sub Salir_Click()
DoCmd.Close
End Sub
