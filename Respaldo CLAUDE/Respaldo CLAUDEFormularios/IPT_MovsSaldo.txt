Private Sub ExisAlm_DblClick(Cancel As Integer)
QueAlm = ExisAlm

If IsNull(QueAlm) = False Then
  If IsNull(QueMod) = False Then
  Dim Dbs As Database
  Dim Rst, Rst1 As Recordset
  Set Dbs = CurrentDb
  Set Rst = Dbs.OpenRecordset("SELECT IdIPT_Modelos FROM IPT_Mod_Alm WHERE IdIPT_Mod_Alm=" & QueMod)
  Set Rst1 = Dbs.OpenRecordset("SELECT IdIPT_Mod_Alm FROM IPT_Mod_Alm WHERE IdIPT_MOdelos=" & Rst!IdIPT_Modelos & " AND IdIPT_Almacenes=" & QueAlm)
QueMod.Requery
QueMod = Rst1!IdIPT_Mod_Alm
Me.Requery
ExisAlm.Requery
Rst.Close
Rst1.Close
Dbs.Close
End If
End If
End Sub
