Private Sub DelAlm_AfterUpdate()

Private Sub AlmSal_AfterUpdate()
AlmEnt.Requery
End Sub

Private Sub HacerTrans_Click()
If IsNull(AlmSal) = True Or IsNull(AlmEnt) = True Then
 MsgBox "Debes de seleccionar un almacen de salida y un almacen de entrada para poder hacer la transferencia entre almacenes", vbInformation, "FALTA DETERMINAR ALMACENES"
Else
DoCmd.OpenForm "IPT_MovTransfer"
Forms![IPT_MovTransfer]![QueAlmE] = AlmEnt
Forms![IPT_MovTransfer]![QueAlmS] = AlmSal

Dim Dbs As Database
Dim Rst, Rst1 As Recordset

Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT Almacen FROM IPT_Almacenes WHERE IdIPT_Almacenes=" & AlmSal)
Set Rst1 = Dbs.OpenRecordset("SELECT Almacen FROM IPT_Almacenes WHERE IdIPT_Almacenes=" & AlmEnt)

Forms![IPT_MovTransfer]![AlmacenSa] = Rst!Almacen
Forms![IPT_MovTransfer]![AlmacenEn] = Rst1!Almacen
Rst.Close
Rst1.Close
Dbs.Close
End If
End Sub

Private Sub QueAlm_DblClick(Cancel As Integer)
If QueTipoMov = 0 Then
 MsgBox "Antes de poder hacer un movimiento es necesario que indiques si es de ENTRADA o de SALIDA", vbInformation, "FALTA ESPECIFICAR MOVIMIENTO"
Else
DoCmd.OpenForm "IPT_Movimientos", , , , acFormAdd
Forms![IPT_Movimientos]![QueIdIPT_Almacenes] = QueAlm
Forms![IPT_Movimientos]![QueEnSa] = QueTipoMov
Forms![IPT_Movimientos]![IdIPT_TipoMov].Requery
  If QueTipoMov = 1 Then
    Forms![IPT_Movimientos]![TituloMov].Caption = "ENTRADA DE ALMACEN"
    Forms![IPT_Movimientos]![TituloMov].BackColor = 65280
  ElseIf QueTipoMov = 2 Then
    Forms![IPT_Movimientos]![TituloMov].Caption = "SALIDA DE ALMACEN"
    Forms![IPT_Movimientos]![TituloMov].BackColor = 255
  End If
End If
End Sub

DoCmd.Close
