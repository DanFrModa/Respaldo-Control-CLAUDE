Private Sub RelacionarOrds_Click()
Dim Dbs As Database
Dim Mismo As Recordset
Dim QueOrd As Recordset
Dim strmensaje As String
Dim Res
Set Dbs = CurrentDb


Set Mismo = Me.RecordsetClone
        Do While Not Mismo.EOF
        
        Set QueOrd = Dbs.OpenRecordset("SELECT Numero, IdOrdenes, IdEtiquetasM, IdClientes, Modelo FROM Ordenes INNER JOIN Modelos ON Ordenes.IdModelos = Modelos.IdModelos WHERE Modelo Like '" & NumMod & "*'")


             If QueOrd.EOF Then
             
             Else
            strmensaje = "Modelo: " & QueOrd!Modelo & ". Orden a Agregar: " & QueOrd!Numero
            Res = MsgBox(strmensaje, vbYesNoCancel, "AGREGAR ORDEN")
            
            If Res = vbCancel Then GoTo OtroMas
            If Res = vbNo Then GoTo ElSig
            If Res = vbYes Then
            IdOrdenes = QueOrd!IdOrdenes
            IdEtiquetasM = QueOrd!IdEtiquetasM
            IdClientes = QueOrd!IdClientes
            End If
            End If
ElSig:
        

          Mismo.MoveNext
          DoCmd.GoToRecord , , acNext
        
        Loop

OtroMas:
QueOrd.Close
Mismo.Close
Dbs.Close
End Sub
