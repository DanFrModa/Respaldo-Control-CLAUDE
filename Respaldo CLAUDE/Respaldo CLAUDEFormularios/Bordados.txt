Option Compare Database

Private Sub Form_Close()
If EstaCargado("Modelos") = True Then
Forms![Modelos]![ModelosBor].Form![IdBordados].Requery
End If
End Sub

Private Sub Form_Open(Cancel As Integer)
DoCmd.GoToRecord acForm, "Bordados", acNewRec
End Sub

Private Sub Foto1_GotFocus()
If IsNull(Me!Foto1) = True Then
Me!Foto1 = Me!Nombre1
End If
End Sub

Private Sub Mismo_Click()
Me!Foto = Me!Nombre
End Sub

Private Sub Salir_Click()
DoCmd.Close
End Sub
Private Sub Agregar_Click()
On Error GoTo Err_Agregar_Click


    DoCmd.GoToRecord , , acNewRec
    Me!Nombre1.SetFocus
Exit_Agregar_Click:
    Exit Sub

Err_Agregar_Click:
    MsgBox Err.Description
    Resume Exit_Agregar_Click
    
End Sub
Private Sub VerFoto_Click()
On Error GoTo Err_VerFoto_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "BordadosFotos"
    
    stLinkCriteria = "[IdBordados]=" & Me![IdBordados]
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_VerFoto_Click:
    Exit Sub

Err_VerFoto_Click:
    MsgBox Err.Description
    Resume Exit_VerFoto_Click
    
End Sub
Sub Cuadro_combinado63_AfterUpdate()
    ' Buscar el registro que coincida con el control.
    Me.RecordsetClone.FindFirst "[IdBordados] = " & Me![Cuadro combinado63]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End Sub
