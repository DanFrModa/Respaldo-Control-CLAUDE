Private Sub Alta_Click()
Dim QFecha As Date
QFecha = QueFecha

End Sub

Private Sub AgregarActividades()
Dim Dbs As Database
Dim Rst As Recordset
Set Dbs = CurrentDb
Set Rst = Dbs.OpenRecordset("SELECT IdIP_Personal, Activo FROM IP_Personal WHERE Activo=True")

Rst.MoveLast
Rst.MoveFirst
  
 Do While Not Rst.EOF
 DoCmd.GoToRecord , , acNewRec
       ' Dbs.Execute "INSERT INTO IP_Productiv (IdIP_Personal,Fecha) VALUES (" & Rst!IdIp_Personal & "," & QFecha & ")"
Me.Fecha = QueFecha
Me.IdIp_Personal = Rst!IdIp_Personal

            Rst.MoveNext
     Loop


Rst.Close
Dbs.Close

End Sub
