Private Sub Form_Current()
If Activo = True Then
Me!Act = "ACTIVO"
Me!Act.BackColor = vbGreen

ElseIf Activo = False Then
Me!Act = "INACTIVO"
Me!Act.BackColor = vbRed
End If
End Sub

Private Sub QueGenero_AfterUpdate()
If IsNull(QueGenero) = True Then
QueGeneroOculto = "*"
Else
QueGeneroOculto = QueGenero
End If
QueModeloLista.Requery
End Sub

Private Sub QueModelo_AfterUpdate()

    Me.RecordsetClone.FindFirst "[IdModelos] = " & Me![QueModelo]
    Me.Bookmark = Me.RecordsetClone.Bookmark
    QueModeloLista = QueModelo
End Sub

End Sub
