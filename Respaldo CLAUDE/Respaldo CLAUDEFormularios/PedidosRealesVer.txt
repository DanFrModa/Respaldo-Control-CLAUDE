Option Compare Database
Option Explicit

Private Sub AgregarPedReal_Click()
If MsgBox(("Con esta accion, se va a generar una Pedido Real relacionado al pedido interno No. " & NumeroPed & " y es imposible borrarlo. Estas seguro de querer meter un nuevo Pedido Real?"), vbYesNo, "Meter Nuevo Pedido Real") = vbYes Then
  If IsNull(IdUsuarioACT) = True Or IdUsuarioACT = 0 Then
   MsgBox "No tienes un usuario valido para meter un nuevo pedido. Salte de CONTROL y vuelvete a meter para continuar", vbExclamation, "USUARIO INVALIDO"
   Exit Sub
  End If
Dim QueIdPR As Integer
Dim Dbs As Database
Dim QueId, Rst As Recordset
Set Dbs = CurrentDb

Dbs.Execute "INSERT INTO PedidosReales (IdPedidos, IdUsuarios, FechaUsuario) VALUES (" & Me.IdPedidos & "," & IdUsuarioACT & ",'" & Now & "')"
Set QueId = Dbs.OpenRecordset("SELECT max(IdPedidosReales) AS UltimoId FROM PedidosReales")
QueId.MoveLast
QueIdPR = QueId!UltimoId
QueId.Close

Set Rst = Dbs.OpenRecordset("SELECT IdPedidosDet FROM PedidosDet WHERE IdPedidos=" & IdPedidos)

Rst.MoveLast
Rst.MoveFirst
  
 Do While Not Rst.EOF
       Dbs.Execute "INSERT INTO PedidosRealesDet (IdPedidosReales, IdPedidosDet) VALUES (" & QueIdPR & "," & Rst!IdPedidosDet & ")"

            Rst.MoveNext
     Loop


Rst.Close
Dbs.Close

DoCmd.OpenForm "PedRealAgregar", , , "[IdPedidosReales]=" & QueIdPR
End If
End Sub

  Forms![PedidosPorMes].Requery

Private Sub Form_Open(Cancel As Integer)
If EstaCargado("PedidosPorMes") = True Then
  QuePed = Forms![PedidosPorMes]![IdPedidos]
 
  QuePR.Requery
    Me.RecordsetClone.FindFirst "[IdPedidos] = " & Me![QuePed]
    Me.Bookmark = Me.RecordsetClone.Bookmark
End If

If PrP(14) = True Then
ModificarPed.Visible = True
End If

End Sub

End Sub
